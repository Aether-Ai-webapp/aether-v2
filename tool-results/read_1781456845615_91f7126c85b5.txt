    24→    24→function getGreeting(): string {
    25→    25→  const hour = new Date().getHours()
    26→    26→  if (hour < 12) return 'Good morning'
    27→    27→  if (hour < 17) return 'Good afternoon'
    28→    28→  return 'Good evening'
    29→    29→}
    30→    30→
    31→    31→function detectContentType(text: string): 'link' | 'task' | 'note' {
    32→    32→  const lower = text.toLowerCase()
    33→    33→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    34→    34→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    35→    35→  return 'note'
    36→    36→}
    37→    37→
    38→    38→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    39→    39→  if (detected === 'link') return 'link'
    40→    40→  return 'text'
    41→    41→}

    62→    62→function generateParticles(count: number = 12): Particle[] {
    63→    63→  const colors = [
    64→    64→    'rgba(168, 85, 247, 0.8)',    // purple-500
    65→    65→    'rgba(192, 132, 252, 0.7)',   // purple-400
    66→    66→    'rgba(139, 92, 246, 0.6)',    // violet-500
    67→    67→    'rgba(124, 58, 237, 0.6)',    // violet-600
    68→    68→    'rgba(109, 40, 217, 0.5)',    // purple-700
    69→    69→  ]
    70→    70→  return Array.from({ length: count }, (_, i) => ({
    71→    71→    id: i,
    72→    72→    x: (Math.random() - 0.5) * 120,
    73→    73→    y: -(60 + Math.random() * 100),
    74→    74→    scale: 0.4 + Math.random() * 0.8,
    75→    75→    rotate: (Math.random() - 0.5) * 360,
    76→    76→    duration: 0.5 + Math.random() * 0.5,
    77→    77→    color: colors[Math.floor(Math.random() * colors.length)],
    78→    78→    size: 4 + Math.random() * 6,
    79→    79→  }))
    80→    80→}

   175→   175→export function Dashboard() {
   176→   176→  const {
   177→   177→    memories,
   178→   178→    saveMemory,
   179→   179→    fetchMemories,
   180→   180→    deleteMemoryFromDB,
   181→   181→    darkMode,
   182→   182→    isLoading,
   183→   183→    isAuthenticated,
   184→   184→    requireAuth,
   185→   185→  } = useAetherStore()
   186→   186→
   187→   187→  const [captureText, setCaptureText] = useState('')
   188→   188→  const [isSaving, setIsSaving] = useState(false)
   189→   189→  const [isJustSaved, setIsJustSaved] = useState(false)
   190→   190→
   191→   191→  // ── Dopamine Engine state
   192→   192→  const [particles, setParticles] = useState<Particle[]>([])
   193→   193→  const [captureFeedback, setCaptureFeedback] = useState<CaptureFeedback | null>(null)
   194→   194→  const [lastSavedId, setLastSavedId] = useState<string | null>(null)
   195→   195→
   196→   196→  const justSavedTimer = useRef<ReturnType<typeof setTimeout>>()
   197→   197→  const feedbackTimer = useRef<ReturnType<typeof setTimeout>>()
   198→   198→
   199→   199→  // Hydration guard
   200→   200→  const [mounted, setMounted] = useState(false)
   201→   201→  useEffect(() => { setMounted(true) }, [])
   202→   202→
   203→   203→  // Skeleton loading
   204→   204→  const [hasFetched, setHasFetched] = useState(false)
   205→   205→
   206→   206→  useEffect(() => {
   207→   207→    if (memories.length > 0 && !hasFetched) setHasFetched(true)
   208→   208→  }, [memories.length, hasFetched])
   209→   209→
   210→   210→  useEffect(() => {
   211→   211→    if (!isLoading && !hasFetched) {
   212→   212→      const timer = setTimeout(() => setHasFetched(true), 600)
   213→   213→      return () => clearTimeout(timer)
   214→   214→    }
   215→   215→  }, [isLoading, hasFetched])
   216→   216→
   217→   217→  const [selectedMemory, setSelectedMemory] = useState<Memory | null>(null)
   218→   218→  const [isDeleting, setIsDeleting] = useState(false)
   219→   219→  const [showPaywall, setShowPaywall] = useState(false)
   220→   220→
   221→   221→  const FREE_MEMORY_LIMIT = 15
   222→   222→
   223→   223→  // ── Capture handler (Dopamine Engine wired in)
   224→   224→  const handleCapture = useCallback(async () => {
   225→   225→    const text = captureText.trim()
   226→   226→    if (!text || isSaving) return
   227→   227→
   228→   228→    // ── AUTH GATE: Block submission if not signed in ──────────────────
   229→   229→    if (!isAuthenticated) {
   230→   230→      // Save the text so we can replay after login
   231→   231→      const pendingText = text
   232→   232→      setCaptureText('')
   233→   233→      requireAuth(async () => {
   234→   234→        // After successful auth, auto-capture the queued thought
   235→   235→        const detectedType = detectContentType(pendingText)
   236→   236→        const memoryType = mapToMemoryType(detectedType)
   237→   237→
   238→   238→        setParticles(generateParticles(14))
   239→   239→        setTimeout(() => setParticles([]), 1200)
   240→   240→        setCaptureFeedback({ category: detectedType })
   241→   241→        if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   242→   242→        feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   243→   243→        setIsJustSaved(true)
   244→   244→        if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   245→   245→        justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   246→   246→
   247→   247→        setIsSaving(true)
   248→   248→        try {
   249→   249→          const savedMemory = await saveMemory({
   250→   250→            type: memoryType,
   251→   251→            title: pendingText.split('\n')[0].slice(0, 80) || 'Quick Note',
   252→   252→            content: pendingText,
   253→   253→            sourceUrl: detectedType === 'link' ? pendingText.trim() : null,
   254→   254→          })
   255→   255→          if (savedMemory) {
   256→   256→            setLastSavedId(savedMemory.id)
   257→   257→            setTimeout(() => setLastSavedId(null), 1500)
   258→   258→          }
   259→   259→        } catch {
   260→   260→          // silent
   261→   261→        } finally {
   262→   262→          setIsSaving(false)
   263→   263→        }
   264→   264→      })
   265→   265→      return
   266→   266→    }
   267→   267→
   268→   268→    const detectedType = detectContentType(text)
   269→   269→    const memoryType = mapToMemoryType(detectedType)
   270→   270→
   271→   271→    setCaptureText('')
   272→   272→    setParticles(generateParticles(14))
   273→   273→    setTimeout(() => setParticles([]), 1200)
   274→   274→
   275→   275→    setCaptureFeedback({ category: detectedType })
   276→   276→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   277→   277→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   278→   278→
   279→   279→    setIsJustSaved(true)
   280→   280→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   281→   281→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   282→   282→
   283→   283→    if (memories.length >= FREE_MEMORY_LIMIT) {
   284→   284→      setTimeout(() => setShowPaywall(true), 800)
   285→   285→      return
   286→   286→    }
   287→   287→
   288→   288→    setIsSaving(true)
   289→   289→    try {
   290→   290→      const savedMemory = await saveMemory({
   291→   291→        type: memoryType,
   292→   292→        title: text.split('\n')[0].slice(0, 80) || 'Quick Note',
   293→   293→        content: text,
   294→   294→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   295→   295→      })
   296→   296→      // NOTE: No fetchMemories() needed — saveMemory already does optimistic addMemory()
   297→   297→      // The store's addMemory() instantly updates the list.
   298→   298→
   299→   299→      if (savedMemory) {
   300→   300→        setLastSavedId(savedMemory.id)
   301→   301→        setTimeout(() => setLastSavedId(null), 1500)
   302→   302→      }
   303→   303→    } catch {
   304→   304→      // silent
   305→   305→    } finally {
   306→   306→      setIsSaving(false)
   307→   307→    }
   308→   308→  }, [captureText, isSaving, saveMemory, memories.length, FREE_MEMORY_LIMIT, isAuthenticated, requireAuth])
   309→   309→
   310→   310→  const handleCaptureKeyDown = useCallback((e: React.KeyboardEvent) => {
   311→   311→    if (e.key === 'Enter' && !e.shiftKey) {
   312→   312→      e.preventDefault()
   313→   313→      handleCapture()
   314→   314→    }
   315→   315→  }, [handleCapture])
   316→   316→
   317→   317→  const handleMicClick = useCallback(() => {
   318→   318→    toast.info('Voice recording coming soon!', { icon: <Mic className="size-4" /> })
   319→   319→  }, [])
   320→   320→
   321→   321→  const handleDeleteMemory = useCallback(async () => {
   322→   322→    if (!selectedMemory || isDeleting) return
   323→   323→    setIsDeleting(true)
   324→   324→    try {
   325→   325→      await deleteMemoryFromDB(selectedMemory.id)
   326→   326→      setSelectedMemory(null)
   327→   327→    } catch {
   328→   328→      // keep modal open
   329→   329→    } finally {
   330→   330→      setIsDeleting(false)
   331→   331→    }
   332→   332→  }, [selectedMemory, isDeleting, deleteMemoryFromDB])
   333→   333→
   334→   334→  const displayMemories = useMemo(() => {
   335→   335→    return [...memories]
   336→   336→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   337→   337→      .slice(0, 5)
   338→   338→  }, [memories])
   339→   339→
   340→   340→  const isDark = darkMode
   341→   341→
   342→   342→  return (
   343→   343→    <motion.div
   344→   344→      initial={{ opacity: 0, y: 20 }}
   345→   345→      animate={{ opacity: 1, y: 0 }}
   346→   346→      transition={{ duration: 0.6, ease: [0.22, 1, 0.36, 1] }}
   347→   347→      className={cn(
   348→   348→        'min-h-screen relative overflow-hidden flex flex-col items-center pt-12 pb-10 px-4',
   349→   349→        isDark ? 'bg-transparent text-white' : 'bg-gradient-to-b from-slate-50 to-white text-gray-900'
   350→   350→      )}
   351→   351→    >
   352→   352→      {/* ── Greeting ──────────────────────────────────────────────── */}
   353→   353→      <section className="relative z-10 w-full max-w-2xl">
   354→   354→        <h1 className={cn(
   355→   355→          'text-3xl font-bold tracking-tight mb-8',
   356→   356→          isDark
   357→   357→            ? 'bg-gradient-to-r from-white via-purple-100 to-violet-200 bg-clip-text text-transparent'
   358→   358→            : 'text-gray-900'
   359→   359→        )}>
   360→   360→          {mounted ? getGreeting() : ''}
   361→   361→        </h1>
   362→   362→      </section>
   363→   363→
   364→   364→      {/* ── Gravity Capture Bar ───────────────────────────────────── */}
   365→   365→      <section className="relative z-10 w-full max-w-2xl mx-auto mb-10">
   366→   366→        <div className={cn(
   367→   367→          'relative rounded-2xl p-1.5 transition-all duration-500',
   368→   368→          isDark
   369→   369→            ? isJustSaved
   370→   370→              ? 'bg-white/[0.03] border border-purple-400/30 shadow-[0_0_0_1px_rgba(168,85,247,0.4),0_0_60px_-10px_rgba(168,85,247,0.4)]'
   371→   371→              : 'bg-white/[0.02] border border-white/[0.04] shadow-[0_0_0_1px_rgba(255,255,255,0.02),0_0_30px_-10px_rgba(168,85,247,0.08)] focus-within:shadow-[0_0_0_1px_rgba(168,85,247,0.2),0_0_60px_-15px_rgba(168,85,247,0.3)] focus-within:border-purple-500/20'
   372→   372→            : isJustSaved
   373→   373→              ? 'bg-white border border-purple-300/50 shadow-[0_0_0_1px_rgba(168,85,247,0.3),0_0_40px_-10px_rgba(168,85,247,0.2)]'

   383→   383→              className={cn(

   385→   385→                isDark
   386→   386→                  ? 'text-white placeholder:text-white/20'

   389→   389→            />
