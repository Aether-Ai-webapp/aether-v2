    28→    28→// ─── Helpers ────────────────────────────────────────────────────────
    29→    29→function getGreeting(): string {
    30→    30→  const hour = new Date().getHours()
    31→    31→  if (hour < 12) return 'Good morning'
    32→    32→  if (hour < 17) return 'Good afternoon'
    33→    33→  return 'Good evening'
    34→    34→}
    35→    35→
    36→    36→function detectContentType(text: string): 'link' | 'task' | 'note' {
    37→    37→  const lower = text.toLowerCase()
    38→    38→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    39→    39→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    40→    40→  return 'note'
    41→    41→}
    42→    42→
    43→    43→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    44→    44→  if (detected === 'link') return 'link'
    45→    45→  return 'text'
    46→    46→}

   176→   176→              toast.error('Could not detect speech')
   177→   177→            }
   178→   178→          } else {
   179→   179→            toast.error('Transcription failed')
   180→   180→          }

   272→   272→          } else {
   273→   273→            toast.error('Failed to save')
   274→   274→          }
   275→   275→        } catch {
   276→   276→          toast.error('Something went wrong')
   277→   277→        } finally {
   278→   278→          setIsSaving(false)
   279→   279→        }
   280→   280→      })

   282→   282→    }
   283→   283→
   284→   284→    const detectedType = detectContentType(text)
   285→   285→    const memoryType = pendingImage ? 'image' : mapToMemoryType(detectedType)
   286→   286→
   287→   287→    setCaptureText('')
   288→   288→    setCaptureFeedback(detectedType)
   289→   289→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   290→   290→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 1800)
   291→   291→
   292→   292→    setIsJustSaved(true)
   293→   293→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   294→   294→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 500)
   295→   295→
   296→   296→    if (memories.length >= FREE_MEMORY_LIMIT) {

   298→   298→      return

   300→   300→
   301→   301→    let imageUrl: string | null = null
   302→   302→    if (pendingImage) {

   304→   304→      removePendingImage()
   305→   305→    }

   307→   307→    setIsSaving(true)

   310→   310→        type: memoryType,
   311→   311→        title: text.split('\n')[0].slice(0, 80) || (imageUrl ? 'Image note' : 'Quick Note'),
   312→   312→        content: text,
   313→   313→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   314→   314→        imageUrl,
   315→   315→      })
   316→   316→
   317→   317→      if (savedMemory) {
   318→   318→        setLastSavedId(savedMemory.id)
   319→   319→        setTimeout(() => setLastSavedId(null), 1200)
   320→   320→        fetchMemories()
   321→   321→      } else {
   322→   322→        toast.error('Failed to save')
   323→   323→      }
   324→   324→    } catch {
   325→   325→      toast.error('Something went wrong')
   326→   326→    } finally {
   327→   327→      setIsSaving(false)

   330→   330→

   386→   386→              initial={{ opacity: 0, y: 6 }}
   387→   387→              animate={{ opacity: 1, y: 0 }}

   392→   392→              <img
   393→   393→                src={pendingImagePreview}
   394→   394→                alt="Preview"

   402→   402→                className="size-5 rounded-md flex items-center justify-center text-zinc-400 hover:text-zinc-600 transition-colors"

   415→   415→              : 'bg-white/60 backdrop-blur-xl border border-black/[0.03] shadow-[0_8px_30px_rgb(0,0,0,0.01)] focus-within:border-purple-300/60 focus-within:shadow-[0_0_50px_rgba(168,85,247,0.05)]'

   417→   417→        >
   418→   418→          <div className="flex items-center gap-2 px-3 py-2.5 min-w-0">
