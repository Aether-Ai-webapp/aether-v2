    23→// ─── Helpers ────────────────────────────────────────────────────────
    24→function getGreeting(): string {
    25→  const hour = new Date().getHours()
    26→  if (hour < 12) return 'Good morning'
    27→  if (hour < 17) return 'Good afternoon'
    28→  return 'Good evening'
    29→}
    30→
    31→function detectContentType(text: string): 'link' | 'task' | 'note' {
    32→  const lower = text.toLowerCase()
    33→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    34→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    35→  return 'note'
    36→}
    37→
    38→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    39→  if (detected === 'link') return 'link'
    40→  return 'text'
    41→}

    50→// ─── Particle Burst Engine ───────────────────────────────────────────
    51→interface Particle {
    52→  id: number
    53→  x: number
    54→  y: number
    55→  scale: number
    56→  rotate: number
    57→  duration: number
    58→  color: string
    59→  size: number
    60→}
    61→
    62→function generateParticles(count: number = 12): Particle[] {
    63→  const colors = [
    64→    'rgba(168, 85, 247, 0.8)',    // purple-500
    65→    'rgba(192, 132, 252, 0.7)',   // purple-400
    66→    'rgba(139, 92, 246, 0.6)',    // violet-500
    67→    'rgba(124, 58, 237, 0.6)',    // violet-600
    68→    'rgba(109, 40, 217, 0.5)',    // purple-700
    69→  ]
    70→  return Array.from({ length: count }, (_, i) => ({
    71→    id: i,
    72→    x: (Math.random() - 0.5) * 120,
    73→    y: -(60 + Math.random() * 100),
    74→    scale: 0.4 + Math.random() * 0.8,
    75→    rotate: (Math.random() - 0.5) * 360,
    76→    duration: 0.5 + Math.random() * 0.5,
    77→    color: colors[Math.floor(Math.random() * colors.length)],
    78→    size: 4 + Math.random() * 6,
    79→  }))
    80→}
    81→
    82→function ParticleBurst({ particles, isDark }: { particles: Particle[]; isDark: boolean }) {
    83→  return (
    84→    <div className="absolute inset-0 pointer-events-none overflow-visible z-20">
    85→      {particles.map((p) => (
    86→        <motion.div
    87→          key={p.id}
    88→          initial={{ opacity: 1, x: 0, y: 0, scale: 0 }}
    89→          animate={{
    90→            opacity: [1, 1, 0],
    91→            x: p.x,
    92→            y: p.y,
    93→            scale: [0, p.scale, 0],
    94→            rotate: p.rotate,
    95→          }}
    96→          transition={{
    97→            duration: p.duration,
    98→            ease: [0.22, 1, 0.36, 1],
    99→          }}
   100→          className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 rounded-full"
   101→          style={{
   102→            width: p.size,
   103→            height: p.size,
   104→            backgroundColor: p.color,
   105→            boxShadow: isDark
   106→              ? `0 0 ${p.size * 2}px ${p.color}`
   107→              : `0 0 ${p.size}px ${p.color}`,
   108→          }}
   109→        />
   110→      ))}
   111→    </div>
   112→  )
   113→}

   174→// ─── Main Dashboard Component ────────────────────────────────────────
   175→export function Dashboard() {
   176→  const {
   177→    memories,
   178→    saveMemory,
   179→    fetchMemories,
   180→    deleteMemoryFromDB,
   181→    darkMode,
   182→    isLoading,
   183→    isAuthenticated,
   184→    requireAuth,
   185→  } = useAetherStore()
   186→
   187→  const [captureText, setCaptureText] = useState('')
   188→  const [isSaving, setIsSaving] = useState(false)
   189→  const [isJustSaved, setIsJustSaved] = useState(false)
   190→
   191→  // ── Dopamine Engine state
   192→  const [particles, setParticles] = useState<Particle[]>([])
   193→  const [captureFeedback, setCaptureFeedback] = useState<CaptureFeedback | null>(null)
   194→  const [lastSavedId, setLastSavedId] = useState<string | null>(null)
   195→
   196→  const justSavedTimer = useRef<ReturnType<typeof setTimeout>>()
   197→  const feedbackTimer = useRef<ReturnType<typeof setTimeout>>()
   198→
   199→  // Hydration guard
   200→  const [mounted, setMounted] = useState(false)
   201→  useEffect(() => { setMounted(true) }, [])
   202→
   203→  // Skeleton loading
   204→  const [hasFetched, setHasFetched] = useState(false)
   205→
   206→  useEffect(() => {
   207→    if (memories.length > 0 && !hasFetched) setHasFetched(true)
   208→  }, [memories.length, hasFetched])
   209→
   210→  useEffect(() => {
   211→    if (!isLoading && !hasFetched) {
   212→      const timer = setTimeout(() => setHasFetched(true), 600)
   213→      return () => clearTimeout(timer)
   214→    }
   215→  }, [isLoading, hasFetched])
   216→
   217→  const [selectedMemory, setSelectedMemory] = useState<Memory | null>(null)
   218→  const [isDeleting, setIsDeleting] = useState(false)
   219→  const [showPaywall, setShowPaywall] = useState(false)
   220→
   221→  const FREE_MEMORY_LIMIT = 15
   222→
   223→  // ── Capture handler (Dopamine Engine wired in)
   224→  const handleCapture = useCallback(async () => {
   225→    const text = captureText.trim()
   226→    if (!text || isSaving) return
   227→
   228→    // ── AUTH GATE: Block submission if not signed in ──────────────────
   229→    if (!isAuthenticated) {
   230→      // Save the text so we can replay after login
   231→      const pendingText = text
   232→      setCaptureText('')
   233→      requireAuth(async () => {
   234→        // After successful auth, auto-capture the queued thought
   235→        const detectedType = detectContentType(pendingText)
   236→        const memoryType = mapToMemoryType(detectedType)
   237→
   238→        setParticles(generateParticles(14))
   239→        setTimeout(() => setParticles([]), 1200)
   240→        setCaptureFeedback({ category: detectedType })
   241→        if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   242→        feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   243→        setIsJustSaved(true)
   244→        if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   245→        justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   246→
   247→        setIsSaving(true)
   248→        try {
   249→          const savedMemory = await saveMemory({
   250→            type: memoryType,
   251→            title: pendingText.split('\n')[0].slice(0, 80) || 'Quick Note',
   252→            content: pendingText,
   253→            sourceUrl: detectedType === 'link' ? pendingText.trim() : null,
   254→          })
   255→          if (savedMemory) {
   256→            setLastSavedId(savedMemory.id)
   257→            setTimeout(() => setLastSavedId(null), 1500)
   258→          }
   259→        } catch {
   260→          // silent
   261→        } finally {
   262→          setIsSaving(false)
   263→        }
   264→      })
   265→      return
   266→    }
   267→
   268→    const detectedType = detectContentType(text)
   269→    const memoryType = mapToMemoryType(detectedType)
   270→
   271→    setCaptureText('')
   272→    setParticles(generateParticles(14))
   273→    setTimeout(() => setParticles([]), 1200)
   274→
   275→    setCaptureFeedback({ category: detectedType })
   276→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   277→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   278→
   279→    setIsJustSaved(true)
   280→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   281→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   282→
   283→    if (memories.length >= FREE_MEMORY_LIMIT) {
   284→      setTimeout(() => setShowPaywall(true), 800)
   285→      return
   286→    }
   287→
   288→    setIsSaving(true)
   289→    try {
   290→      const savedMemory = await saveMemory({
   291→        type: memoryType,
   292→        title: text.split('\n')[0].slice(0, 80) || 'Quick Note',
   293→        content: text,
   294→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   295→      })
   296→      // NOTE: No fetchMemories() needed — saveMemory already does optimistic addMemory()
   297→      // The store's addMemory() instantly updates the list.
   298→
   299→      if (savedMemory) {
   300→        setLastSavedId(savedMemory.id)
   301→        setTimeout(() => setLastSavedId(null), 1500)
   302→      }
   303→    } catch {
   304→      // silent
   305→    } finally {
   306→      setIsSaving(false)
   307→    }
   308→  }, [captureText, isSaving, saveMemory, memories.length, FREE_MEMORY_LIMIT, isAuthenticated, requireAuth])
   309→
   310→  const handleCaptureKeyDown = useCallback((e: React.KeyboardEvent) => {
   311→    if (e.key === 'Enter' && !e.shiftKey) {
   312→      e.preventDefault()
   313→      handleCapture()
   314→    }
   315→  }, [handleCapture])
   316→
   317→  const handleMicClick = useCallback(() => {
   318→    toast.info('Voice recording coming soon!', { icon: <Mic className="size-4" /> })
   319→  }, [])
   320→
   321→  const handleDeleteMemory = useCallback(async () => {
   322→    if (!selectedMemory || isDeleting) return
   323→    setIsDeleting(true)
   324→    try {
   325→      await deleteMemoryFromDB(selectedMemory.id)
   326→      setSelectedMemory(null)
   327→    } catch {
   328→      // keep modal open
   329→    } finally {
   330→      setIsDeleting(false)
   331→    }
   332→  }, [selectedMemory, isDeleting, deleteMemoryFromDB])
   333→
   334→  const displayMemories = useMemo(() => {
   335→    return [...memories]
   336→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   337→      .slice(0, 5)
   338→  }, [memories])
   339→
   340→  const isDark = darkMode
   341→
   342→  return (
   343→    <motion.div
   344→      initial={{ opacity: 0, y: 20 }}
   345→      animate={{ opacity: 1, y: 0 }}
   346→      transition={{ duration: 0.6, ease: [0.22, 1, 0.36, 1] }}
   347→      className={cn(
   348→        'min-h-screen relative overflow-hidden flex flex-col items-center pt-12 pb-10 px-4',
   349→        isDark ? 'bg-transparent text-white' : 'bg-gradient-to-b from-slate-50 to-white text-gray-900'
   350→      )}
   351→    >
   352→      {/* ── Greeting ──────────────────────────────────────────────── */}
   353→      <section className="relative z-10 w-full max-w-2xl">
   354→        <h1 className={cn(
   355→          'text-3xl font-bold tracking-tight mb-8',
   356→          isDark
   357→            ? 'bg-gradient-to-r from-white via-purple-100 to-violet-200 bg-clip-text text-transparent'
   358→            : 'text-gray-900'
   359→        )}>
   360→          {mounted ? getGreeting() : ''}
   361→        </h1>
   362→      </section>
   363→
   364→      {/* ── Gravity Capture Bar ───────────────────────────────────── */}
   365→      <section className="relative z-10 w-full max-w-2xl mx-auto mb-10">
   366→        <div className={cn(
   367→          'relative rounded-2xl p-1.5 transition-all duration-500',
   368→          isDark
   369→            ? isJustSaved
   370→              ? 'bg-white/[0.03] border border-purple-400/30 shadow-[0_0_0_1px_rgba(168,85,247,0.4),0_0_60px_-10px_rgba(168,85,247,0.4)]'
   371→              : 'bg-white/[0.02] border border-white/[0.04] shadow-[0_0_0_1px_rgba(255,255,255,0.02),0_0_30px_-10px_rgba(168,85,247,0.08)] focus-within:shadow-[0_0_0_1px_rgba(168,85,247,0.2),0_0_60px_-15px_rgba(168,85,247,0.3)] focus-within:border-purple-500/20'
   372→            : isJustSaved
   373→              ? 'bg-white border border-purple-300/50 shadow-[0_0_0_1px_rgba(168,85,247,0.3),0_0_40px_-10px_rgba(168,85,247,0.2)]'
   374→              : 'bg-white border border-gray-200 shadow-[0_0_0_1px_rgba(0,0,0,0.03),0_0_30px_-10px_rgba(168,85,247,0.06)] focus-within:shadow-[0_0_0_1px_rgba(168,85,247,0.2),0_0_40px_-10px_rgba(168,85,247,0.1)] focus-within:border-purple-300/30'
   375→        )}>
   376→          <div className="flex items-center gap-2">
   377→            <input
   378→              value={captureText}
   379→              onChange={(e) => setCaptureText(e.target.value)}
   380→              onKeyDown={handleCaptureKeyDown}
   381→              placeholder="Dump a thought, URL, or task... press Enter"
   382→              disabled={isSaving}
   383→              className={cn(
   384→                'w-full bg-transparent text-base focus:outline-none px-4 py-3',
   385→                isDark
   386→                  ? 'text-white placeholder:text-white/20'
   387→                  : 'text-gray-900 placeholder:text-gray-400'
   388→              )}
   389→            />
   390→
   391→            {/* Mic button */}
   392→            <button
   393→              onClick={handleMicClick}
   394→              className={cn(
   395→                'flex items-center justify-center size-9 rounded-lg transition-colors duration-150 shrink-0',
   396→                isDark
   397→                  ? 'text-white/15 hover:text-white/40 hover:bg-white/[0.03]'
   398→                  : 'text-gray-300 hover:text-gray-500 hover:bg-gray-100'
   399→              )}
   400→              aria-label="Voice recording"
   401→            >
   402→              <Mic className="size-4" />
   403→            </button>
   404→
   405→            {/* Save/Send button */}
   406→            <motion.button
   407→              onClick={handleCapture}
   408→              disabled={!captureText.trim() || isSaving}
   409→              whileTap={{ scale: 0.9 }}
   410→              transition={{ type: 'spring', stiffness: 500, damping: 15 }}
   411→              className={cn(
   412→                'flex items-center justify-center text-sm font-medium px-4 py-1.5 rounded-lg transition-all duration-200 shrink-0',
   413→                captureText.trim() && !isSaving
   414→                  ? isDark
   415→                    ? 'bg-gradient-to-r from-purple-400 to-violet-500 hover:from-purple-300 hover:to-violet-400 text-white shadow-[0_0_16px_-4px_rgba(168,85,247,0.5)] hover:shadow-[0_0_24px_-4px_rgba(168,85,247,0.6)]'
   416→                    : 'bg-gradient-to-r from-purple-500 to-violet-500 hover:from-purple-400 hover:to-violet-400 text-white shadow-[0_0_16px_-4px_rgba(168,85,247,0.3)] hover:shadow-[0_0_24px_-4px_rgba(168,85,247,0.5)]'
   417→                  : isDark
   418→                    ? 'bg-white/[0.04] text-white/20'
   419→                    : 'bg-gray-100 text-gray-400'
   420→              )}
   421→            >
   422→              <Send className="size-4" />
   423→            </motion.button>
   424→          </div>
   425→
   426→          {/* 🫧 Particle Burst Layer */}
   427→          <AnimatePresence>
   428→            {particles.length > 0 && (
   429→              <ParticleBurst particles={particles} isDark={isDark} />
   430→            )}
   431→          </AnimatePresence>
   432→        </div>
   433→
   434→        {/* 🎯 Capture Feedback Card */}
   435→        <div className="flex justify-center">
   436→          <AnimatePresence>
   437→            {captureFeedback && (
   438→              <CaptureFeedbackCard feedback={captureFeedback} isDark={isDark} />
   439→            )}
   440→          </AnimatePresence>
   441→        </div>
   442→
   443→        {/* Free Plan Limit */}
   444→        {memories.length > 0 && (
   445→          <div className={cn(
   446→            'flex items-center justify-center gap-1.5 mt-3',
   447→            isDark ? 'text-white/15' : 'text-gray-400'
   448→          )}>
   449→            {memories.length >= FREE_MEMORY_LIMIT ? (
   450→              <>
   451→                <Crown className="size-3 text-purple-500/60" />
   452→                <span className={cn('text-[11px] font-medium', isDark ? 'text-purple-400/60' : 'text-purple-600')}>
   453→                  Free limit reached — upgrade for unlimited
   454→                </span>
   455→              </>
   456→            ) : memories.length >= FREE_MEMORY_LIMIT - 3 ? (
   457→              <>
   458→                <Crown className="size-3" />
   459→                <span className="text-[11px]">
   460→                  {FREE_MEMORY_LIMIT - memories.length} memories remaining on Free plan
   461→                </span>
   462→              </>
   463→            ) : (
   464→              <span className="text-[11px]">
   465→                {memories.length} / {FREE_MEMORY_LIMIT} free memories
   466→              </span>
   467→            )}
   468→          </div>
   469→        )}
   470→      </section>
   471→
   472→      {/* ── Memories Feed ──────────────────────────────────────────── */}
   473→      <section className="relative z-10 w-full max-w-2xl mx-auto space-y-3">
   474→        {!hasFetched ? (
   475→          <>
   476→            {[0, 1, 2].map((i) => (
   477→              <div
   478→                key={i}
   479→                className={cn(
   480→                  'rounded-xl p-5 animate-pulse',
   481→                  isDark
   482→                    ? 'bg-white/[0.015] border border-white/[0.03]'
   483→                    : 'bg-gray-50 border border-gray-100'
   484→                )}
   485→              >
   486→                <div className="flex items-start gap-4">
   487→                  <div className={cn(
   488→                    'size-9 rounded-lg shrink-0',
   489→                    isDark ? 'bg-white/[0.03]' : 'bg-gray-100'
   490→                  )} />
   491→                  <div className="flex-1 space-y-3">
   492→                    <div className={cn('h-4 rounded w-3/4', isDark ? 'bg-white/[0.03]' : 'bg-gray-100')} />
   493→                    <div className={cn('h-3 rounded w-1/2', isDark ? 'bg-white/[0.03]' : 'bg-gray-100')} />
   494→                  </div>
   495→                </div>
   496→              </div>
   497→            ))}
   498→          </>
   499→        ) : displayMemories.length === 0 ? (
   500→          <p className={cn('text-sm mt-20 text-center', isDark ? 'text-white/10' : 'text-gray-300')}>
   501→            Your mind is clear. Dump a thought above.
   502→          </p>
   503→        ) : (
   504→          <>
   505→            {displayMemories.map((memory) => (
   506→              <MemoryCard
   507→                key={memory.id}
   508→                memory={memory}
   509→                isDark={isDark}
   510→                isNew={memory.id === lastSavedId}
   511→                onClick={() => setSelectedMemory(memory)}
   512→              />
   513→            ))}
   514→          </>
   515→        )}

   619→function MemoryDetailModal({
   620→  memory,
   621→  isDark,

   624→  isDeleting,

   626→  memory: Memory
   627→  isDark: boolean
   628→  onClose: () => void

   630→  isDeleting: boolean
   631→}) {
   632→  const [relatedMemories, setRelatedMemories] = useState<Array<{
   633→    id: string
   634→    title: string
   635→    content: string
   636→    tags: string[]

   638→    createdAt: string
   639→    similarity: number | null

   645→  useEffect(() => {

   648→      .then((res) => res.ok ? res.json() : { related: [] })
   649→      .then((data) => setRelatedMemories(data.related || []))
   650→      .catch(() => setRelatedMemories([]))
   651→      .finally(() => setLoadingRelated(false))

   655→  const handleGenerateSummary = useCallback(async () => {
   656→    setGeneratingSummary(true)

   658→      const res = await fetch('/api/ai/summary', {
   659→        method: 'POST',
   660→        headers: { 'Content-Type': 'application/json' },
   661→        body: JSON.stringify({ content: memory.content.slice(0, 1000) }),

   665→        if (summary?.trim()) {
   666→          // Update the memory in the store

   692→        </div>
   693→        ${memory.summary ? `
   694→        <div style="background: #faf5ff; border-left: 3px solid #7c3aed; padding: 16px 20px; margin-bottom: 24px; border-radius: 0 8px 8px 0;">
   695→          <p style="font-size: 11px; font-weight: 700; text-transform: uppercase; letter-spacing: 1px; color: #7c3aed; margin: 0 0 8px 0;">AI Summary</p>
