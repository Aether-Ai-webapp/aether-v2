    75→function mapSupabaseMemory(row: SupabaseMemoryRow): Memory {
    76→  return {
    77→    id: row.id,
    78→    type: (row.type as MemoryType) || 'text',
    79→    title: row.title || '',
    80→    content: row.content || '',
    81→    summary: row.summary,
    82→    tags: row.tags ? row.tags.split(',').filter(Boolean) : [],
    83→    sourceUrl: row.source_url,
    84→    fileUrl: row.file_url,
    85→    imagePreview: row.image_preview,
    86→    isFavorite: row.is_favorite || false,
    87→    createdAt: row.created_at,
    88→    updatedAt: row.updated_at,
    89→    collections: row.memory_collections?.map((mc) => ({

   109→// ─── Smart Token Saver: Local, free, instant tagging ──────────────────
   110→function getLocalTags(text: string): string[] {
   111→  const lower = text.toLowerCase()
   112→  const tags: string[] = []
   113→
   114→  // URL detection → 'link'
   115→  if (/https?:\/\//.test(lower)) {
   116→    tags.push('link')
   117→  }
   118→
   119→  // Task words → 'task'
   120→  if (/\b(todo|remind|need to|must|buy)\b/.test(lower)) {
   121→    tags.push('task')
   122→  }
   123→
   124→  // Person/personal words → 'personal'
   125→  if (/\b(i |me |with |call|email)\b/.test(lower)) {
   126→    tags.push('personal')
   127→  }
   128→
   129→  // Question/idea words → 'idea'
   130→  if (/\b(what|how|why|who)\b/.test(lower)) {
   131→    tags.push('idea')
   132→  }
   133→
   134→  return tags.slice(0, 5)
   135→}

   137→// ─── Extended keyword-based auto-tagging (richer, for AI fallback) ───
   138→function autoGenerateTags(content: string, title: string): string[] {
   139→  const text = `${title} ${content}`.toLowerCase()
   140→  const tagMap: Record<string, string[]> = {
   141→    'work': ['meeting', 'project', 'deadline', 'client', 'office', 'team', 'q1', 'q2', 'q3', 'q4', 'quarterly', 'strategy'],
   142→    'personal': ['routine', 'morning', 'exercise', 'meditation', 'journal', 'habit'],
   143→    'travel': ['trip', 'itinerary', 'flight', 'hotel', 'visit', 'tokyo', 'paris', 'destination'],
   144→    'learning': ['learn', 'study', 'course', 'tutorial', 'book', 'read', 'article'],
   145→    'code': ['code', 'programming', 'react', 'javascript', 'typescript', 'api', 'bug', 'feature', 'css', 'html', 'framework'],
   146→    'design': ['design', 'ui', 'ux', 'layout', 'color', 'font', 'figma', 'wireframe'],
   147→    'ai': ['ai', 'machine learning', 'neural', 'model', 'gpt', 'gemini', 'llm', 'chatbot'],
   148→    'recipe': ['recipe', 'cook', 'bake', 'ingredient', 'food', 'meal', 'breakfast', 'dinner'],
   149→    'idea': ['idea', 'concept', 'brainstorm', 'innovative', 'startup', 'prototype'],
   150→    'finance': ['budget', 'expense', 'invest', 'savings', 'money', 'cost'],
   151→  }
   152→
   153→  const tags: string[] = []
   154→  for (const [tag, keywords] of Object.entries(tagMap)) {
   155→    if (keywords.some(kw => text.includes(kw))) {
   156→      tags.push(tag)
   157→    }
   158→  }
   159→  return tags.slice(0, 5)
   160→}

   257→export const useAetherStore = create<AetherState>((set, get) => ({
   258→  // Navigation
   259→  currentView: 'dashboard',
   260→  setCurrentView: (view) => set({ currentView: view, selectedMemoryId: null, selectedCollectionId: null }),
   261→
   262→  // Auth — start as "local" user, no auth gate
   263→  user: { id: 'local', email: '', name: 'Aether User', avatarUrl: null },
   264→  isAuthenticated: false,
   265→  showAuthModal: false,
   266→  pendingAction: null,
   267→  supabaseReady: false,
   268→  setUser: (user) => set({ user, isAuthenticated: !!user && user.id !== 'local' }),
   269→  setShowAuthModal: (showAuthModal) => set({ showAuthModal }),
   270→
   271→  // The gate: if user is signed in, run action immediately. Otherwise, queue it and show modal.
   272→  requireAuth: (action) => {
   273→    const state = get()
   274→    if (state.isAuthenticated && state.user && state.user.id !== 'local') {
   275→      action()
   276→    } else {
   277→      set({ showAuthModal: true, pendingAction: action })
   278→    }
   279→  },
   280→
   281→  login: async (email, password) => {
   282→    try {
   283→      const res = await fetch('/api/auth/login', {
   284→        method: 'POST',
   285→        headers: { 'Content-Type': 'application/json' },
   286→        body: JSON.stringify({ email, password }),
   287→      })
   288→      if (!res.ok) return false
   289→      const data = await res.json()
   290→      if (data.user) {
   291→        set({
   292→          user: {
   293→            id: data.user.id,
   294→            email: data.user.email || '',
   295→            name: data.user.user_metadata?.name || '',
   296→            avatarUrl: data.user.user_metadata?.avatar_url || null,
   297→          },
   298→          isAuthenticated: true,
   299→          showAuthModal: false,
   300→        })
   301→        // CRITICAL: Await Supabase tables check before running pending actions or fetching data
   302→        // Without this, supabaseReady is still false when saveMemory/fetchMemories run
   303→        await get().checkSupabaseTables()
   304→        // Execute any pending action that was queued (e.g., deferred capture)
   305→        const pending = get().pendingAction
   306→        if (pending) {
   307→          await pending()
   308→          set({ pendingAction: null })
   309→        }
   310→        // Re-fetch data from Supabase (now that supabaseReady is set)
   311→        await get().fetchMemories()
   312→        await get().fetchCollections()
   313→        return true
   314→      }
   315→      return false
   316→    } catch {
   317→      return false
   318→    }
   319→  },
   320→
   321→  signup: async (email, password, name) => {
   322→    try {
   323→      const res = await fetch('/api/auth/signup', {
   324→        method: 'POST',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify({ email, password, name }),
   327→      })
   328→      if (!res.ok) return false
   329→      const data = await res.json()
   330→      if (data.user) {
   331→        set({
   332→          user: {
   333→            id: data.user.id,
   334→            email: data.user.email || '',
   335→            name: name || '',
   336→            avatarUrl: null,
   337→          },
   338→          isAuthenticated: true,
   339→          showAuthModal: false,
   340→        })
   341→        // CRITICAL: Await Supabase tables check before running pending actions or fetching data
   342→        await get().checkSupabaseTables()
   343→        // Execute any pending action
   344→        const pending = get().pendingAction
   345→        if (pending) {
   346→          await pending()
   347→          set({ pendingAction: null })
   348→        }
   349→        // Re-fetch data from Supabase (now that supabaseReady is set)
   350→        await get().fetchMemories()
   351→        await get().fetchCollections()
   352→        return true
   353→      }
   354→      return false
   355→    } catch {
   356→      return false
   357→    }
   358→  },
   359→
   360→  logout: async () => {
   361→    try {
   362→      await fetch('/api/auth/logout', { method: 'POST' })
   363→    } catch { /* ignore */ }
   364→    resetSupabaseBrowser()
   365→    set({
   366→      user: { id: 'local', email: '', name: 'Aether User', avatarUrl: null },
   367→      isAuthenticated: false,
   368→      supabaseReady: false,
   369→      chatMessages: [],
   370→    })
   371→    // Reload data as local user
   372→    get().fetchMemories()
   373→    get().fetchCollections()
   374→  },
   375→
   376→  checkSession: async () => {
   377→    try {
   378→      const res = await fetch('/api/auth/session')
   379→      if (res.ok) {
   380→        const data = await res.json()
   381→        if (data.authenticated && data.user) {
   382→          set({ user: data.user, isAuthenticated: true })
   383→          // CRITICAL: Await Supabase tables check, THEN fetch data
   384→          // Without this, fetchMemories runs while supabaseReady is still false
   385→          await get().checkSupabaseTables()
   386→          // Now fetch data with correct supabaseReady state
   387→          await Promise.all([
   388→            get().fetchMemories(),
   389→            get().fetchCollections(),
   390→          ])
   391→          return
   392→        }
   393→      }
   394→    } catch { /* ignore */ }
   395→    // No Supabase session — stay as local user (not gated)
   396→    set({ isAuthenticated: false })
   397→    // Still try to fetch from Prisma API for local data
   398→    await Promise.all([
   399→      get().fetchMemories(),
   400→      get().fetchCollections(),
   401→    ])
   402→  },
   403→
   404→  // Check if Supabase tables exist for the current user
   405→  checkSupabaseTables: async () => {
   406→    try {
   407→      const supabase = await getSupabaseBrowser()
   408→      const { data: { session } } = await supabase.auth.getSession()
   409→      if (!session) {
   410→        set({ supabaseReady: false })
   411→        return false
   412→      }
   413→
   414→      // Try a minimal query to see if the memories table exists AND RLS lets us read
   415→      const { error } = await supabase
   416→        .from('memories')
   417→        .select('id')
   418→        .eq('user_id', session.user.id)
   419→        .limit(1)
   420→
   421→      if (error) {
   422→        console.warn('[checkSupabaseTables] Probe failed:', error.message)
   423→        // Don't immediately give up — a permissions error on an empty table is normal
   424→        // If the error is specifically "no rows" that's fine, the table exists
   425→        // Only set supabaseReady=false for actual table-not-exist errors
   426→        const isTableMissing = error.message?.includes('does not exist') || error.message?.includes('not found')
   427→        if (!isTableMissing) {
   428→          // Table exists, we just can't read it (could be empty + RLS)
   429→          // This is actually fine — set ready=true
   430→          console.info('[checkSupabaseTables] Table exists but query returned error (likely empty table + RLS). Setting ready=true.')
   431→          set({ supabaseReady: true })
   432→          return true
   433→        }
   434→        set({ supabaseReady: false })
   435→        return false
   436→      }
   437→
   438→      set({ supabaseReady: true })
   439→      return true
   440→    } catch (err) {
   441→      console.error('[checkSupabaseTables] Exception:', err instanceof Error ? err.message : 'Unknown')
   442→      set({ supabaseReady: false })
   443→      return false
   444→    }
   445→  },
   446→
   447→  // Data
   448→  memories: [],
   449→  collections: [],
   450→  isLoading: false,
   451→
   452→  // Selected
   453→  selectedMemoryId: null,
   454→  selectedCollectionId: null,
   455→
   456→  // Search
   457→  searchQuery: '',
   458→  setSearchQuery: (q) => set({ searchQuery: q }),
   459→  filterType: 'all',
   460→  setFilterType: (t) => set({ filterType: t }),
   461→
   462→  // Chat
   463→  chatMessages: [],
   464→  addChatMessage: (msg) => set((s) => ({ chatMessages: [...s.chatMessages, msg] })),
   465→  clearChat: () => set({ chatMessages: [] }),
   466→
   467→  // Theme
   468→  darkMode: false,
   469→  toggleDarkMode: () => set((s) => ({ darkMode: !s.darkMode })),
   470→
   471→  // Data actions
   472→  setMemories: (memories) => set({ memories }),
   473→  setCollections: (collections) => set({ collections }),
   474→  setLoading: (isLoading) => set({ isLoading }),
   475→  setSelectedMemoryId: (id) => set({ selectedMemoryId: id }),
   476→  setSelectedCollectionId: (id) => set({ selectedCollectionId: id }),
   477→  addMemory: (memory) => set((s) => ({ memories: [memory, ...s.memories] })),
   478→  updateMemory: (id, updates) => set((s) => ({
   479→    memories: s.memories.map((m) => (m.id === id ? { ...m, ...updates } : m)),
   480→  })),
   481→  deleteMemory: (id) => set((s) => ({
   482→    memories: s.memories.filter((m) => m.id !== id),
   483→  })),
   484→  addCollection: (collection) => set((s) => ({
   485→    collections: [...s.collections, collection],
   486→  })),
   487→  updateCollection: (id, updates) => set((s) => ({
   488→    collections: s.collections.map((c) => (c.id === id ? { ...c, ...updates } : c)),
   489→  })),
   490→  deleteCollection: (id) => set((s) => ({
   491→    collections: s.collections.filter((c) => c.id !== id),
   492→  })),
   493→
   494→  // ── Fetch Memories ──────────────────────────────────────────────────
   495→  fetchMemories: async () => {
   496→    set({ isLoading: true })
   497→    const state = get()
   498→
   499→    // ── Authenticated + Supabase ready: fetch from Supabase directly ──
   500→    if (state.isAuthenticated && state.supabaseReady) {
   501→      try {
   502→        const supabase = await getSupabaseBrowser()
   503→        const { data: { session } } = await supabase.auth.getSession()
   504→        if (!session) throw new Error('No session')
   505→
   506→        const { data, error } = await supabase
   507→          .from('memories')
   508→          .select('*, memory_collections(collection_id, collections(id, name, color, icon))')
   509→          .eq('user_id', session.user.id)
   510→          .order('created_at', { ascending: false })
   511→
   512→        if (error) throw error
   513→
   514→        const memories = (data as SupabaseMemoryRow[]).map(mapSupabaseMemory)
   515→        set({ memories, isLoading: false })
   516→        return
   517→      } catch (supabaseErr) {
   518→        console.error('[fetchMemories] Supabase direct fetch failed:', supabaseErr instanceof Error ? supabaseErr.message : 'Unknown')
   519→        // Fall through to API
   520→      }
   521→    }
   522→
   523→    // ── Fallback: fetch from API (works for both auth and non-auth) ──
   524→    try {
   525→      const res = await fetch('/api/memories')
   526→      if (res.ok) {
   527→        const memories = await res.json()
   528→        set({ memories, isLoading: false })
   529→        return
   530→      }
   531→    } catch (e) {
   532→      console.error('[fetchMemories] API fetch failed:', e)
   533→    }
   534→    set({ isLoading: false })
   535→  },
   536→
   537→  // ── Fetch Collections ───────────────────────────────────────────────
   538→  fetchCollections: async () => {
   539→    const state = get()
   540→    if (state.isAuthenticated && state.supabaseReady) {
   541→      try {
   542→        const supabase = await getSupabaseBrowser()
   543→        const { data: { session } } = await supabase.auth.getSession()
   544→        if (!session) throw new Error('No session')
   545→
   546→        const { data, error } = await supabase
   547→          .from('collections')
   548→          .select('*, memory_collections(id)')
   549→          .eq('user_id', session.user.id)
   550→          .order('name', { ascending: true })
   551→
   552→        if (error) throw error
   553→
   554→        const collections = (data as SupabaseCollectionRow[]).map(mapSupabaseCollection)
   555→        set({ collections })
   556→        return
   557→      } catch {
   558→        // Fall through to Prisma API
   559→      }
   560→    }
   561→
   562→    // Fallback: fetch from Prisma API
   563→    try {
   564→      const res = await fetch('/api/collections')
   565→      if (res.ok) {
   566→        const collections = await res.json()
   567→        set({ collections })
   568→      }
   569→    } catch (e) {
   570→      console.error('Failed to fetch collections:', e)
   571→    }
   572→  },
   573→
   574→  // ── Delete Memory from DB (Supabase or Prisma) ─────────────────────
   575→  deleteMemoryFromDB: async (id: string) => {
   576→    const state = get()
   577→
   578→    // Authenticated + Supabase: delete from Supabase
   579→    if (state.isAuthenticated && state.supabaseReady) {
   580→      try {
   581→        const supabase = await getSupabaseBrowser()
   582→        const { data: { session } } = await supabase.auth.getSession()
   583→        if (session) {
   584→          await supabase.from('memories').delete().eq('id', id)
   585→        }
   586→      } catch (err) {
   587→        console.error('[deleteMemoryFromDB] Supabase delete failed:', err instanceof Error ? err.message : 'Unknown')
   588→        // Fall through to Prisma API
   589→      }
   590→    }
   591→
