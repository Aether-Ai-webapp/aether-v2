   115→   115→// ─── Smart Token Saver: Local, free, instant tagging ──────────────────
   116→   116→function getLocalTags(text: string): string[] {
   117→   117→  const lower = text.toLowerCase()
   118→   118→  const tags: string[] = []
   119→   119→
   120→   120→  // URL detection → 'link'
   121→   121→  if (/https?:\/\//.test(lower)) {
   122→   122→    tags.push('link')
   123→   123→  }
   124→   124→
   125→   125→  // Task words → 'task'
   126→   126→  if (/\b(todo|remind|need to|must|buy)\b/.test(lower)) {
   127→   127→    tags.push('task')
   128→   128→  }
   129→   129→
   130→   130→  // Person/personal words → 'personal'
   131→   131→  if (/\b(i |me |with |call|email)\b/.test(lower)) {
   132→   132→    tags.push('personal')
   133→   133→  }
   134→   134→
   135→   135→  // Question/idea words → 'idea'
   136→   136→  if (/\b(what|how|why|who)\b/.test(lower)) {
   137→   137→    tags.push('idea')
   138→   138→  }
   139→   139→
   140→   140→  return tags.slice(0, 5)
   141→   141→}

   143→   143→// ─── Extended keyword-based auto-tagging (richer, for AI fallback) ───
   144→   144→function autoGenerateTags(content: string, title: string): string[] {
   145→   145→  const text = `${title} ${content}`.toLowerCase()
   146→   146→  const tagMap: Record<string, string[]> = {
   147→   147→    'work': ['meeting', 'project', 'deadline', 'client', 'office', 'team', 'q1', 'q2', 'q3', 'q4', 'quarterly', 'strategy'],
   148→   148→    'personal': ['routine', 'morning', 'exercise', 'meditation', 'journal', 'habit'],
   149→   149→    'travel': ['trip', 'itinerary', 'flight', 'hotel', 'visit', 'tokyo', 'paris', 'destination'],
   150→   150→    'learning': ['learn', 'study', 'course', 'tutorial', 'book', 'read', 'article'],
   151→   151→    'code': ['code', 'programming', 'react', 'javascript', 'typescript', 'api', 'bug', 'feature', 'css', 'html', 'framework'],
   152→   152→    'design': ['design', 'ui', 'ux', 'layout', 'color', 'font', 'figma', 'wireframe'],
   153→   153→    'ai': ['ai', 'machine learning', 'neural', 'model', 'gpt', 'gemini', 'llm', 'chatbot'],
   154→   154→    'recipe': ['recipe', 'cook', 'bake', 'ingredient', 'food', 'meal', 'breakfast', 'dinner'],
   155→   155→    'idea': ['idea', 'concept', 'brainstorm', 'innovative', 'startup', 'prototype'],
   156→   156→    'finance': ['budget', 'expense', 'invest', 'savings', 'money', 'cost'],
   157→   157→  }
   158→   158→
   159→   159→  const tags: string[] = []
   160→   160→  for (const [tag, keywords] of Object.entries(tagMap)) {
   161→   161→    if (keywords.some(kw => text.includes(kw))) {
   162→   162→      tags.push(tag)
   163→   163→    }
   164→   164→  }
   165→   165→  return tags.slice(0, 5)
   166→   166→}

   290→   290→  login: async (email, password) => {

   293→   293→        method: 'POST',

   295→   295→        body: JSON.stringify({ email, password }),
   296→   296→      })

   304→   304→            name: data.user.user_metadata?.name || '',
   305→   305→            avatarUrl: data.user.user_metadata?.avatar_url || null,
   306→   306→          },
   307→   307→          isAuthenticated: true,

   311→   311→        // Without this, supabaseReady is still false when saveMemory/fetchMemories run
   312→   312→        await get().checkSupabaseTables()
   313→   313→        // Execute any pending action that was queued (e.g., deferred capture)

   315→   315→        if (pending) {
   316→   316→          await pending()
   317→   317→          set({ pendingAction: null })
   318→   318→        }
   319→   319→        // Re-fetch data from Supabase (now that supabaseReady is set)
   320→   320→        await get().fetchMemories()
   321→   321→        await get().fetchCollections()
   322→   322→        return true
   323→   323→      }
   324→   324→      return false
   325→   325→    } catch {
   326→   326→      return false
   327→   327→    }

   409→   409→      get().fetchCollections(),

   416→   416→      const supabase = await getSupabaseBrowser()
   417→   417→      const { data: { session } } = await supabase.auth.getSession()
   418→   418→      if (!session) {
   419→   419→        set({ supabaseReady: false })
   420→   420→        return false
   421→   421→      }
   422→   422→
   423→   423→      // Try a minimal query to see if the memories table exists AND RLS lets us read
   424→   424→      const { error } = await supabase
   425→   425→        .from('memories')
   426→   426→        .select('id')
   427→   427→        .eq('user_id', session.user.id)
   428→   428→        .limit(1)
   429→   429→
   430→   430→      if (error) {
   431→   431→        console.warn('[checkSupabaseTables] Probe failed:', error.message)
   432→   432→        // Don't immediately give up — a permissions error on an empty table is normal
   433→   433→        // If the error is specifically "no rows" that's fine, the table exists
   434→   434→        // Only set supabaseReady=false for actual table-not-exist errors
   435→   435→        const isTableMissing = error.message?.includes('does not exist') || error.message?.includes('not found')
   436→   436→        if (!isTableMissing) {
   437→   437→          // Table exists, we just can't read it (could be empty + RLS)
   438→   438→          // This is actually fine — set ready=true
   439→   439→          console.info('[checkSupabaseTables] Table exists but query returned error (likely empty table + RLS). Setting ready=true.')
   440→   440→          set({ supabaseReady: true })
   441→   441→          return true
   442→   442→        }
   443→   443→        set({ supabaseReady: false })
   444→   444→        return false
   445→   445→      }
   446→   446→
   447→   447→      set({ supabaseReady: true })
   448→   448→      return true
   449→   449→    } catch (err) {
   450→   450→      console.error('[checkSupabaseTables] Exception:', err instanceof Error ? err.message : 'Unknown')
   451→   451→      set({ supabaseReady: false })
   452→   452→      return false
   453→   453→    }
   454→   454→  },
