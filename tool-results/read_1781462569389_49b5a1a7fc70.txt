    24→    24→function getGreeting(): string {
    25→    25→  const hour = new Date().getHours()
    26→    26→  if (hour < 12) return 'Good morning'
    27→    27→  if (hour < 17) return 'Good afternoon'
    28→    28→  return 'Good evening'
    29→    29→}
    30→    30→
    31→    31→function detectContentType(text: string): 'link' | 'task' | 'note' {
    32→    32→  const lower = text.toLowerCase()
    33→    33→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    34→    34→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    35→    35→  return 'note'
    36→    36→}
    37→    37→
    38→    38→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    39→    39→  if (detected === 'link') return 'link'
    40→    40→  return 'text'
    41→    41→}

    62→    62→function generateParticles(count: number = 12): Particle[] {
    63→    63→  const colors = [
    64→    64→    'rgba(124, 58, 237, 0.8)',    // violet-600
    65→    65→    'rgba(99, 102, 241, 0.7)',    // indigo-500
    66→    66→    'rgba(79, 70, 229, 0.6)',     // indigo-600
    67→    67→    'rgba(59, 130, 246, 0.5)',    // blue-500
    68→    68→    'rgba(37, 99, 235, 0.5)',     // blue-600
    69→    69→  ]
    70→    70→  return Array.from({ length: count }, (_, i) => ({
    71→    71→    id: i,
    72→    72→    x: (Math.random() - 0.5) * 120,
    73→    73→    y: -(60 + Math.random() * 100),
    74→    74→    scale: 0.4 + Math.random() * 0.8,
    75→    75→    rotate: (Math.random() - 0.5) * 360,
    76→    76→    duration: 0.5 + Math.random() * 0.5,
    77→    77→    color: colors[Math.floor(Math.random() * colors.length)],
    78→    78→    size: 4 + Math.random() * 6,
    79→    79→  }))
    80→    80→}

   175→   175→export function Dashboard() {
   176→   176→  const {
   177→   177→    memories,
   178→   178→    saveMemory,
   179→   179→    fetchMemories,
   180→   180→    deleteMemoryFromDB,
   181→   181→    darkMode,
   182→   182→    isLoading,
   183→   183→    isAuthenticated,
   184→   184→    requireAuth,
   185→   185→  } = useAetherStore()
   186→   186→
   187→   187→  const [captureText, setCaptureText] = useState('')
   188→   188→  const [isSaving, setIsSaving] = useState(false)
   189→   189→  const [isJustSaved, setIsJustSaved] = useState(false)
   190→   190→
   191→   191→  // ── Dopamine Engine state
   192→   192→  const [particles, setParticles] = useState<Particle[]>([])
   193→   193→  const [captureFeedback, setCaptureFeedback] = useState<CaptureFeedback | null>(null)
   194→   194→  const [lastSavedId, setLastSavedId] = useState<string | null>(null)
   195→   195→
   196→   196→  const justSavedTimer = useRef<ReturnType<typeof setTimeout>>()
   197→   197→  const feedbackTimer = useRef<ReturnType<typeof setTimeout>>()
   198→   198→
   199→   199→  // Hydration guard
   200→   200→  const [mounted, setMounted] = useState(false)
   201→   201→  useEffect(() => { setMounted(true) }, [])
   202→   202→
   203→   203→  // Skeleton loading
   204→   204→  const [hasFetched, setHasFetched] = useState(false)
   205→   205→
   206→   206→  useEffect(() => {
   207→   207→    if (memories.length > 0 && !hasFetched) setHasFetched(true)
   208→   208→  }, [memories.length, hasFetched])
   209→   209→
   210→   210→  useEffect(() => {
   211→   211→    if (!isLoading && !hasFetched) {
   212→   212→      const timer = setTimeout(() => setHasFetched(true), 600)
   213→   213→      return () => clearTimeout(timer)
   214→   214→    }
   215→   215→  }, [isLoading, hasFetched])
   216→   216→
   217→   217→  const [selectedMemory, setSelectedMemory] = useState<Memory | null>(null)
   218→   218→  const [isDeleting, setIsDeleting] = useState(false)
   219→   219→  const [showPaywall, setShowPaywall] = useState(false)
   220→   220→
   221→   221→  const FREE_MEMORY_LIMIT = 15
   222→   222→
   223→   223→  // ── Capture handler (Dopamine Engine wired in)
   224→   224→  const handleCapture = useCallback(async () => {
   225→   225→    const text = captureText.trim()
   226→   226→    if (!text || isSaving) return
   227→   227→
   228→   228→    // ── AUTH GATE: Block submission if not signed in ──────────────────
   229→   229→    if (!isAuthenticated) {
   230→   230→      // Save the text so we can replay after login
   231→   231→      const pendingText = text
   232→   232→      setCaptureText('')
   233→   233→      requireAuth(async () => {
   234→   234→        // After successful auth, auto-capture the queued thought
   235→   235→        const detectedType = detectContentType(pendingText)
   236→   236→        const memoryType = mapToMemoryType(detectedType)
   237→   237→
   238→   238→        setParticles(generateParticles(14))
   239→   239→        setTimeout(() => setParticles([]), 1200)
   240→   240→        setCaptureFeedback({ category: detectedType })
   241→   241→        if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   242→   242→        feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   243→   243→        setIsJustSaved(true)
   244→   244→        if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   245→   245→        justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   246→   246→
   247→   247→        setIsSaving(true)
   248→   248→        try {
   249→   249→          const savedMemory = await saveMemory({
   250→   250→            type: memoryType,
   251→   251→            title: pendingText.split('\n')[0].slice(0, 80) || 'Quick Note',
   252→   252→            content: pendingText,
   253→   253→            sourceUrl: detectedType === 'link' ? pendingText.trim() : null,
   254→   254→          })
   255→   255→          if (savedMemory) {
   256→   256→            setLastSavedId(savedMemory.id)
   257→   257→            setTimeout(() => setLastSavedId(null), 1500)
   258→   258→            // GUARANTEED HYDRATION after auth-gate replay
   259→   259→            fetchMemories()
   260→   260→          } else {
   261→   261→            toast.error('Failed to save. Please try again.')
   262→   262→          }
   263→   263→        } catch {
   264→   264→          toast.error('Something went wrong while saving.')
   265→   265→        } finally {
   266→   266→          setIsSaving(false)
   267→   267→        }
   268→   268→      })
   269→   269→      return
   270→   270→    }
   271→   271→
   272→   272→    const detectedType = detectContentType(text)
   273→   273→    const memoryType = mapToMemoryType(detectedType)
   274→   274→
   275→   275→    setCaptureText('')
   276→   276→    setParticles(generateParticles(14))
   277→   277→    setTimeout(() => setParticles([]), 1200)
   278→   278→
   279→   279→    setCaptureFeedback({ category: detectedType })
   280→   280→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   281→   281→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   282→   282→
   283→   283→    setIsJustSaved(true)
   284→   284→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   285→   285→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   286→   286→
   287→   287→    if (memories.length >= FREE_MEMORY_LIMIT) {
   288→   288→      setTimeout(() => setShowPaywall(true), 800)
   289→   289→      return
   290→   290→    }
   291→   291→
   292→   292→    setIsSaving(true)
   293→   293→    try {
   294→   294→      const savedMemory = await saveMemory({
   295→   295→        type: memoryType,
   296→   296→        title: text.split('\n')[0].slice(0, 80) || 'Quick Note',
   297→   297→        content: text,
   298→   298→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   299→   299→      })
   300→   300→
   301→   301→      if (savedMemory) {
   302→   302→        setLastSavedId(savedMemory.id)
   303→   303→        setTimeout(() => setLastSavedId(null), 1500)
   304→   304→        // GUARANTEED HYDRATION: Re-fetch from backend to confirm persistence
   305→   305→        // The store's saveMemory() does optimistic addMemory(), but this
   306→   306→        // confirms the data actually made it to Supabase/Prisma
   307→   307→        fetchMemories()
   308→   308→      } else {
   309→   309→        // saveMemory returned null — both Supabase and Prisma failed
   310→   310→        toast.error('Failed to save. Please try again.')
   311→   311→      }
   312→   312→    } catch {
   313→   313→      toast.error('Something went wrong while saving.')
   314→   314→    } finally {
   315→   315→      setIsSaving(false)
   316→   316→    }
   317→   317→  }, [captureText, isSaving, saveMemory, fetchMemories, memories.length, FREE_MEMORY_LIMIT, isAuthenticated, requireAuth])
   318→   318→
   319→   319→  const handleCaptureKeyDown = useCallback((e: React.KeyboardEvent) => {
   320→   320→    if (e.key === 'Enter' && !e.shiftKey) {
   321→   321→      e.preventDefault()
   322→   322→      handleCapture()
   323→   323→    }
   324→   324→  }, [handleCapture])
   325→   325→
   326→   326→  const handleMicClick = useCallback(() => {
   327→   327→    toast.info('Voice recording coming soon!', { icon: <Mic className="size-4" /> })
   328→   328→  }, [])
   329→   329→
   330→   330→  const handleDeleteMemory = useCallback(async () => {
   331→   331→    if (!selectedMemory || isDeleting) return
   332→   332→    setIsDeleting(true)
   333→   333→    try {
   334→   334→      await deleteMemoryFromDB(selectedMemory.id)
   335→   335→      setSelectedMemory(null)
   336→   336→    } catch {
   337→   337→      // keep modal open
   338→   338→    } finally {
   339→   339→      setIsDeleting(false)
   340→   340→    }
   341→   341→  }, [selectedMemory, isDeleting, deleteMemoryFromDB])
   342→   342→
   343→   343→  const displayMemories = useMemo(() => {
   344→   344→    return [...memories]
   345→   345→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())

   347→   347→  }, [memories])
   348→   348→
   349→   349→  const isDark = darkMode
   350→   350→
   351→   351→  return (
   352→   352→    <motion.div
   353→   353→      initial={{ opacity: 0, y: 20 }}
   354→   354→      animate={{ opacity: 1, y: 0 }}
   355→   355→      transition={{ duration: 0.6, ease: [0.22, 1, 0.36, 1] }}
   356→   356→      className={cn(
   357→   357→        'min-h-screen relative overflow-hidden flex flex-col items-center pt-12 pb-10 px-4',
   358→   358→        isDark ? 'bg-transparent text-zinc-100' : 'bg-gradient-to-b from-slate-50 to-white text-gray-900'
   359→   359→      )}
   360→   360→    >
   361→   361→      {/* ── Greeting ──────────────────────────────────────────────── */}
   362→   362→      <section className="relative z-10 w-full max-w-2xl">
   363→   363→        <h1 className={cn(
   364→   364→          'text-3xl font-bold tracking-tight mb-8',
   365→   365→          isDark
   366→   366→            ? 'bg-gradient-to-b from-white to-zinc-400 bg-clip-text text-transparent'
   367→   367→            : 'text-gray-900'
   368→   368→        )}>
   369→   369→          {mounted ? getGreeting() : ''}
   370→   370→        </h1>
   371→   371→      </section>
   372→   372→
   373→   373→      {/* ── Gravity Capture Bar ───────────────────────────────────── */}
   374→   374→      <section className="relative z-10 w-full max-w-2xl mx-auto mb-10">
   375→   375→        <div className={cn(
   376→   376→          'relative rounded-2xl p-1.5 transition-all duration-500',
   377→   377→          isDark
   378→   378→            ? isJustSaved
   379→   379→              ? 'bg-[#16171B]/80 backdrop-blur-md border border-purple-500/30 shadow-[0_0_0_1px_rgba(168,85,247,0.4),0_0_40px_-8px_rgba(168,85,247,0.35)]'
   380→   380→              : 'bg-[#16171B]/80 backdrop-blur-md border border-white/[0.06] shadow-inner shadow-black/20 focus-within:ring-1 focus-within:ring-purple-500/50 focus-within:shadow-[0_0_30px_rgba(168,85,247,0.15)] focus-within:border-purple-500/20'
   381→   381→            : isJustSaved
   382→   382→              ? 'bg-white border border-purple-300/50 shadow-[0_0_0_1px_rgba(168,85,247,0.3),0_0_40px_-10px_rgba(168,85,247,0.2)]'
   383→   383→              : 'bg-white border border-gray-200 shadow-[0_0_0_1px_rgba(0,0,0,0.03),0_0_30px_-10px_rgba(168,85,247,0.06)] focus-within:shadow-[0_0_0_1px_rgba(168,85,247,0.2),0_0_40px_-10px_rgba(168,85,247,0.1)] focus-within:border-purple-300/30'

   385→   385→          <div className="flex items-center gap-2">
   386→   386→            <input

   388→   388→              onChange={(e) => setCaptureText(e.target.value)}

   422→   422→                captureText.trim() && !isSaving
