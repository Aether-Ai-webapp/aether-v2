    23→// ─── Helpers ────────────────────────────────────────────────────────
    24→function getGreeting(): string {
    25→  const hour = new Date().getHours()
    26→  if (hour < 12) return 'Good morning'
    27→  if (hour < 17) return 'Good afternoon'
    28→  return 'Good evening'
    29→}
    30→
    31→function detectContentType(text: string): 'link' | 'task' | 'note' {
    32→  const lower = text.toLowerCase()
    33→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    34→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    35→  return 'note'
    36→}
    37→
    38→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    39→  if (detected === 'link') return 'link'
    40→  return 'text'
    41→}

    50→// ─── Particle Burst Engine ───────────────────────────────────────────
    51→interface Particle {
    52→  id: number
    53→  x: number
    54→  y: number
    55→  scale: number
    56→  rotate: number
    57→  duration: number
    58→  color: string
    59→  size: number
    60→}
    61→
    62→function generateParticles(count: number = 12): Particle[] {
    63→  const colors = [
    64→    'rgba(124, 58, 237, 0.8)',    // violet-600
    65→    'rgba(99, 102, 241, 0.7)',    // indigo-500
    66→    'rgba(79, 70, 229, 0.6)',     // indigo-600
    67→    'rgba(59, 130, 246, 0.5)',    // blue-500
    68→    'rgba(37, 99, 235, 0.5)',     // blue-600
    69→  ]
    70→  return Array.from({ length: count }, (_, i) => ({
    71→    id: i,
    72→    x: (Math.random() - 0.5) * 120,
    73→    y: -(60 + Math.random() * 100),
    74→    scale: 0.4 + Math.random() * 0.8,
    75→    rotate: (Math.random() - 0.5) * 360,
    76→    duration: 0.5 + Math.random() * 0.5,
    77→    color: colors[Math.floor(Math.random() * colors.length)],
    78→    size: 4 + Math.random() * 6,
    79→  }))
    80→}
    81→
    82→function ParticleBurst({ particles, isDark }: { particles: Particle[]; isDark: boolean }) {
    83→  return (
    84→    <div className="absolute inset-0 pointer-events-none overflow-visible z-20">
    85→      {particles.map((p) => (
    86→        <motion.div
    87→          key={p.id}
    88→          initial={{ opacity: 1, x: 0, y: 0, scale: 0 }}
    89→          animate={{
    90→            opacity: [1, 1, 0],
    91→            x: p.x,
    92→            y: p.y,
    93→            scale: [0, p.scale, 0],
    94→            rotate: p.rotate,
    95→          }}
    96→          transition={{
    97→            duration: p.duration,
    98→            ease: [0.22, 1, 0.36, 1],
    99→          }}
   100→          className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 rounded-full"
   101→          style={{
   102→            width: p.size,
   103→            height: p.size,
   104→            backgroundColor: p.color,
   105→            boxShadow: isDark
   106→              ? `0 0 ${p.size * 2}px ${p.color}`
   107→              : `0 0 ${p.size}px ${p.color}`,
   108→          }}
   109→        />
   110→      ))}
   111→    </div>
   112→  )
   113→}

   174→// ─── Main Dashboard Component ────────────────────────────────────────
   175→export function Dashboard() {
   176→  const {
   177→    memories,
   178→    saveMemory,
   179→    fetchMemories,
   180→    deleteMemoryFromDB,
   181→    darkMode,
   182→    isLoading,
   183→    isAuthenticated,
   184→    requireAuth,
   185→  } = useAetherStore()
   186→
   187→  const [captureText, setCaptureText] = useState('')
   188→  const [isSaving, setIsSaving] = useState(false)
   189→  const [isJustSaved, setIsJustSaved] = useState(false)
   190→
   191→  // ── Dopamine Engine state
   192→  const [particles, setParticles] = useState<Particle[]>([])
   193→  const [captureFeedback, setCaptureFeedback] = useState<CaptureFeedback | null>(null)
   194→  const [lastSavedId, setLastSavedId] = useState<string | null>(null)
   195→
   196→  const justSavedTimer = useRef<ReturnType<typeof setTimeout>>()
   197→  const feedbackTimer = useRef<ReturnType<typeof setTimeout>>()
   198→
   199→  // Hydration guard
   200→  const [mounted, setMounted] = useState(false)
   201→  useEffect(() => { setMounted(true) }, [])
   202→
   203→  // Skeleton loading
   204→  const [hasFetched, setHasFetched] = useState(false)
   205→
   206→  useEffect(() => {
   207→    if (memories.length > 0 && !hasFetched) setHasFetched(true)
   208→  }, [memories.length, hasFetched])
   209→
   210→  useEffect(() => {
   211→    if (!isLoading && !hasFetched) {
   212→      const timer = setTimeout(() => setHasFetched(true), 600)
   213→      return () => clearTimeout(timer)
   214→    }
   215→  }, [isLoading, hasFetched])
   216→
   217→  const [selectedMemory, setSelectedMemory] = useState<Memory | null>(null)
   218→  const [isDeleting, setIsDeleting] = useState(false)
   219→  const [showPaywall, setShowPaywall] = useState(false)
   220→
   221→  const FREE_MEMORY_LIMIT = 15
   222→
   223→  // ── Capture handler (Dopamine Engine wired in)
   224→  const handleCapture = useCallback(async () => {
   225→    const text = captureText.trim()
   226→    if (!text || isSaving) return
   227→
   228→    // ── AUTH GATE: Block submission if not signed in ──────────────────
   229→    if (!isAuthenticated) {
   230→      // Save the text so we can replay after login
   231→      const pendingText = text
   232→      setCaptureText('')
   233→      requireAuth(async () => {
   234→        // After successful auth, auto-capture the queued thought
   235→        const detectedType = detectContentType(pendingText)
   236→        const memoryType = mapToMemoryType(detectedType)
   237→
   238→        setParticles(generateParticles(14))
   239→        setTimeout(() => setParticles([]), 1200)
   240→        setCaptureFeedback({ category: detectedType })
   241→        if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   242→        feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   243→        setIsJustSaved(true)
   244→        if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   245→        justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   246→
   247→        setIsSaving(true)
   248→        try {
   249→          const savedMemory = await saveMemory({
   250→            type: memoryType,
   251→            title: pendingText.split('\n')[0].slice(0, 80) || 'Quick Note',
   252→            content: pendingText,
   253→            sourceUrl: detectedType === 'link' ? pendingText.trim() : null,
   254→          })
   255→          if (savedMemory) {
   256→            setLastSavedId(savedMemory.id)
   257→            setTimeout(() => setLastSavedId(null), 1500)
   258→            // GUARANTEED HYDRATION after auth-gate replay
   259→            fetchMemories()
   260→          } else {
   261→            toast.error('Failed to save. Please try again.')
   262→          }
   263→        } catch {
   264→          toast.error('Something went wrong while saving.')
   265→        } finally {
   266→          setIsSaving(false)
   267→        }
   268→      })
   269→      return
   270→    }
   271→
   272→    const detectedType = detectContentType(text)
   273→    const memoryType = mapToMemoryType(detectedType)
   274→
   275→    setCaptureText('')
   276→    setParticles(generateParticles(14))
   277→    setTimeout(() => setParticles([]), 1200)
   278→
   279→    setCaptureFeedback({ category: detectedType })
   280→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   281→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   282→
   283→    setIsJustSaved(true)
   284→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   285→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   286→
   287→    if (memories.length >= FREE_MEMORY_LIMIT) {
   288→      setTimeout(() => setShowPaywall(true), 800)
   289→      return
   290→    }
   291→
   292→    setIsSaving(true)
   293→    try {
   294→      const savedMemory = await saveMemory({
   295→        type: memoryType,
   296→        title: text.split('\n')[0].slice(0, 80) || 'Quick Note',
   297→        content: text,
   298→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   299→      })
   300→
   301→      if (savedMemory) {
   302→        setLastSavedId(savedMemory.id)
   303→        setTimeout(() => setLastSavedId(null), 1500)
   304→        // GUARANTEED HYDRATION: Re-fetch from backend to confirm persistence
   305→        // The store's saveMemory() does optimistic addMemory(), but this
   306→        // confirms the data actually made it to Supabase/Prisma
   307→        fetchMemories()
   308→      } else {
   309→        // saveMemory returned null — both Supabase and Prisma failed
   310→        toast.error('Failed to save. Please try again.')
   311→      }
   312→    } catch {
   313→      toast.error('Something went wrong while saving.')
   314→    } finally {
   315→      setIsSaving(false)
   316→    }
   317→  }, [captureText, isSaving, saveMemory, fetchMemories, memories.length, FREE_MEMORY_LIMIT, isAuthenticated, requireAuth])
   318→
   319→  const handleCaptureKeyDown = useCallback((e: React.KeyboardEvent) => {
   320→    if (e.key === 'Enter' && !e.shiftKey) {
   321→      e.preventDefault()
   322→      handleCapture()
   323→    }
   324→  }, [handleCapture])
   325→
   326→  const handleMicClick = useCallback(() => {
   327→    toast.info('Voice recording coming soon!', { icon: <Mic className="size-4" /> })
   328→  }, [])
   329→
   330→  const handleDeleteMemory = useCallback(async () => {
   331→    if (!selectedMemory || isDeleting) return
   332→    setIsDeleting(true)
   333→    try {
   334→      await deleteMemoryFromDB(selectedMemory.id)
   335→      setSelectedMemory(null)
   336→    } catch {
   337→      // keep modal open
   338→    } finally {
   339→      setIsDeleting(false)
   340→    }
   341→  }, [selectedMemory, isDeleting, deleteMemoryFromDB])
   342→
   343→  const displayMemories = useMemo(() => {
   344→    return [...memories]
   345→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   346→      .slice(0, 5)
   347→  }, [memories])
   348→
   349→  const isDark = darkMode
   350→
   351→  return (
   352→    <motion.div
   353→      initial={{ opacity: 0, y: 20 }}
   354→      animate={{ opacity: 1, y: 0 }}
   355→      transition={{ duration: 0.6, ease: [0.22, 1, 0.36, 1] }}
   356→      className={cn(
   357→        'min-h-screen relative overflow-hidden flex flex-col items-center pt-12 pb-10 px-4',
   358→        isDark ? 'bg-transparent text-zinc-100' : 'bg-gradient-to-b from-slate-50 to-white text-gray-900'
   359→      )}
   360→    >
   361→      {/* ── Greeting ──────────────────────────────────────────────── */}
   362→      <section className="relative z-10 w-full max-w-2xl">
   363→        <h1 className={cn(
   364→          'text-3xl font-bold tracking-tight mb-8',
   365→          isDark
   366→            ? 'bg-gradient-to-b from-white to-zinc-400 bg-clip-text text-transparent'
   367→            : 'text-gray-900'
   368→        )}>
   369→          {mounted ? getGreeting() : ''}
   370→        </h1>
   371→      </section>
   372→
   373→      {/* ── Gravity Capture Bar ───────────────────────────────────── */}
   374→      <section className="relative z-10 w-full max-w-2xl mx-auto mb-10">
   375→        <div className={cn(
   376→          'relative rounded-2xl p-1.5 transition-all duration-500',
   377→          isDark
   378→            ? isJustSaved
   379→              ? 'bg-[#16171B]/80 backdrop-blur-md border border-purple-500/30 shadow-[0_0_0_1px_rgba(168,85,247,0.4),0_0_40px_-8px_rgba(168,85,247,0.35)]'
   380→              : 'bg-[#16171B]/80 backdrop-blur-md border border-white/[0.06] shadow-inner shadow-black/20 focus-within:ring-1 focus-within:ring-purple-500/50 focus-within:shadow-[0_0_30px_rgba(168,85,247,0.15)] focus-within:border-purple-500/20'
   381→            : isJustSaved
   382→              ? 'bg-white border border-purple-300/50 shadow-[0_0_0_1px_rgba(168,85,247,0.3),0_0_40px_-10px_rgba(168,85,247,0.2)]'
   383→              : 'bg-white border border-gray-200 shadow-[0_0_0_1px_rgba(0,0,0,0.03),0_0_30px_-10px_rgba(168,85,247,0.06)] focus-within:shadow-[0_0_0_1px_rgba(168,85,247,0.2),0_0_40px_-10px_rgba(168,85,247,0.1)] focus-within:border-purple-300/30'
   384→        )}>
   385→          <div className="flex items-center gap-2">
   386→            <input
   387→              value={captureText}
   388→              onChange={(e) => setCaptureText(e.target.value)}
   389→              onKeyDown={handleCaptureKeyDown}
   390→              placeholder="Dump a thought, URL, or task... press Enter"
   391→              disabled={isSaving}
   392→              className={cn(
   393→                'w-full bg-transparent text-base focus:outline-none px-4 py-3',
   394→                isDark
   395→                  ? 'text-zinc-100 placeholder:text-zinc-500'
   396→                  : 'text-gray-900 placeholder:text-gray-400'
   397→              )}
   398→            />
   399→
   400→            {/* Mic button */}
   401→            <button
   402→              onClick={handleMicClick}
   403→              className={cn(
   404→                'flex items-center justify-center size-9 rounded-lg transition-colors duration-150 shrink-0',
   405→                isDark
   406→                  ? 'text-zinc-500 hover:text-zinc-300 hover:bg-zinc-800/40'
   407→                  : 'text-gray-300 hover:text-gray-500 hover:bg-gray-100'
   408→              )}
   409→              aria-label="Voice recording"
   410→            >
   411→              <Mic className="size-4" />
   412→            </button>
   413→
   414→            {/* Save/Send button */}
   415→            <motion.button
   416→              onClick={handleCapture}
   417→              disabled={!captureText.trim() || isSaving}
   418→              whileTap={{ scale: 0.9 }}
   419→              transition={{ type: 'spring', stiffness: 500, damping: 15 }}
   420→              className={cn(
   421→                'flex items-center justify-center text-sm font-medium px-4 py-1.5 rounded-lg transition-all duration-200 shrink-0',
   422→                captureText.trim() && !isSaving
   423→                  ? isDark
   424→                    ? 'bg-gradient-to-r from-purple-600 via-indigo-500 to-blue-500 hover:from-purple-500 hover:via-indigo-400 hover:to-blue-400 text-white shadow-[0_0_16px_-4px_rgba(168,85,247,0.5)] hover:shadow-[0_0_24px_-4px_rgba(168,85,247,0.6)]'
   425→                    : 'bg-gradient-to-r from-purple-500 to-violet-500 hover:from-purple-400 hover:to-violet-400 text-white shadow-[0_0_16px_-4px_rgba(168,85,247,0.3)] hover:shadow-[0_0_24px_-4px_rgba(168,85,247,0.5)]'
   426→                  : isDark
   427→                    ? 'bg-zinc-800/40 text-zinc-600'
   428→                    : 'bg-gray-100 text-gray-400'
   429→              )}
   430→            >
   431→              <Send className="size-4" />
   432→            </motion.button>
   433→          </div>
   434→
   435→          {/* 🫧 Particle Burst Layer */}
   436→          <AnimatePresence>
   437→            {particles.length > 0 && (
   438→              <ParticleBurst particles={particles} isDark={isDark} />
   439→            )}
   440→          </AnimatePresence>
   441→        </div>
   442→
   443→        {/* 🎯 Capture Feedback Card */}
   444→        <div className="flex justify-center">
   445→          <AnimatePresence>
   446→            {captureFeedback && (
   447→              <CaptureFeedbackCard feedback={captureFeedback} isDark={isDark} />
   448→            )}
   449→          </AnimatePresence>
   450→        </div>
   451→
   452→        {/* Free Plan Limit */}
   453→        {memories.length > 0 && (
   454→          <div className={cn(
   455→            'flex items-center justify-center gap-1.5 mt-3',
   456→            isDark ? 'text-zinc-500' : 'text-gray-400'
   457→          )}>
   458→            {memories.length >= FREE_MEMORY_LIMIT ? (
   459→              <>
   460→                <Crown className="size-3 text-purple-500/60" />
   461→                <span className={cn('text-[11px] font-medium', isDark ? 'text-indigo-400/60' : 'text-purple-600')}>
   462→                  Free limit reached — upgrade for unlimited
   463→                </span>
   464→              </>
   465→            ) : memories.length >= FREE_MEMORY_LIMIT - 3 ? (
   466→              <>
   467→                <Crown className="size-3" />
   468→                <span className="text-[11px]">
   469→                  {FREE_MEMORY_LIMIT - memories.length} memories remaining on Free plan
   470→                </span>
   471→              </>
   472→            ) : (
   473→              <span className="text-[11px]">
   474→                {memories.length} / {FREE_MEMORY_LIMIT} free memories
   475→              </span>
   476→            )}
   477→          </div>
   478→        )}
   479→      </section>
   480→
   481→      {/* ── Memories Feed ──────────────────────────────────────────── */}
   482→      <section className="relative z-10 w-full max-w-2xl mx-auto space-y-3">
   483→        {!hasFetched ? (
   484→          <>
   485→            {[0, 1, 2].map((i) => (
   486→              <div
   487→                key={i}
   488→                className={cn(
   489→                  'rounded-xl p-5 animate-pulse',
   490→                  isDark
   491→                    ? 'bg-zinc-800/30 border border-white/[0.04]'
   492→                    : 'bg-gray-50 border border-gray-100'
   493→                )}
   494→              >
   495→                <div className="flex items-start gap-4">
   496→                  <div className={cn(
   497→                    'size-9 rounded-lg shrink-0',
   498→                    isDark ? 'bg-zinc-700/20' : 'bg-gray-100'
   499→                  )} />
   500→                  <div className="flex-1 space-y-3">
   501→                    <div className={cn('h-4 rounded w-3/4', isDark ? 'bg-zinc-700/20' : 'bg-gray-100')} />
   502→                    <div className={cn('h-3 rounded w-1/2', isDark ? 'bg-zinc-700/20' : 'bg-gray-100')} />
   503→                  </div>
   504→                </div>
   505→              </div>
   506→            ))}
   507→          </>
   508→        ) : displayMemories.length === 0 ? (
   509→          <p className={cn('text-sm mt-20 text-center', isDark ? 'text-zinc-600' : 'text-gray-300')}>
