   471→// ─── Main Memories Component ────────────────────────────────────────
   472→export function Memories() {
   473→  const {
   474→    memories,
   475→    collections,
   476→    searchQuery,
   477→    setSearchQuery,
   478→    filterType,
   479→    setFilterType,
   480→    selectedMemoryId,
   481→    setSelectedMemoryId,
   482→    selectedCollectionId,
   483→    setSelectedCollectionId,
   484→    deleteMemory,
   485→    updateMemory,
   486→    setCurrentView,
   487→  } = useAetherStore()
   488→
   489→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null)
   490→  const [isDeleting, setIsDeleting] = useState(false)
   491→
   492→  // ── Filtered memories ────────────────────────────────────────────
   493→  const filteredMemories = useMemo(() => {
   494→    let result = [...memories]
   495→
   496→    // Filter by selected collection
   497→    if (selectedCollectionId) {
   498→      result = result.filter((m) =>
   499→        m.collections.some((c) => c.id === selectedCollectionId)
   500→      )
   501→    }
   502→
   503→    // Filter by type
   504→    if (filterType !== 'all') {
   505→      result = result.filter((m) => m.type === filterType)
   506→    }
   507→
   508→    // Search filter
   509→    if (searchQuery.trim()) {
   510→      const q = searchQuery.toLowerCase().trim()
   511→      result = result.filter(
   512→        (m) =>
   513→          m.title.toLowerCase().includes(q) ||
   514→          m.content.toLowerCase().includes(q) ||
   515→          m.tags.some((t) => t.toLowerCase().includes(q))
   516→      )
   517→    }
   518→
   519→    // Sort by date (newest first)
   520→    result.sort(
   521→      (a, b) =>
   522→        new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   523→    )
   524→
   525→    return result
   526→  }, [memories, searchQuery, filterType, selectedCollectionId])
   527→
   528→  // ── Selected memory ──────────────────────────────────────────────
   529→  const selectedMemory = useMemo(
   530→    () => memories.find((m) => m.id === selectedMemoryId) || null,
   531→    [memories, selectedMemoryId]
   532→  )
   533→
   534→  // ── Selected collection name ─────────────────────────────────────
   535→  const selectedCollection = useMemo(
   536→    () => collections.find((c) => c.id === selectedCollectionId) || null,
   537→    [collections, selectedCollectionId]
   538→  )
   539→
   540→  // ── Handlers ─────────────────────────────────────────────────────
   541→  const handleToggleFavorite = useCallback(
   542→    async (memory: Memory) => {
   543→      const newFavorite = !memory.isFavorite
   544→      // Optimistic update
   545→      updateMemory(memory.id, { isFavorite: newFavorite })
   546→      try {
   547→        await fetch(`/api/memories/${memory.id}`, {
   548→          method: 'PATCH',
   549→          headers: { 'Content-Type': 'application/json' },
   550→          body: JSON.stringify({ isFavorite: newFavorite }),
   551→        })
   552→      } catch {
   553→        // Revert on error
   554→        updateMemory(memory.id, { isFavorite: !newFavorite })
   555→      }
   556→    },
   557→    [updateMemory]
   558→  )
   559→
   560→  const handleDelete = useCallback(async () => {
   561→    if (!deleteTarget) return
   562→    setIsDeleting(true)
   563→    try {
   564→      const res = await fetch(`/api/memories/${deleteTarget}`, {
   565→        method: 'DELETE',
   566→      })
   567→      if (res.ok) {
   568→        deleteMemory(deleteTarget)
   569→        if (selectedMemoryId === deleteTarget) {
   570→          setSelectedMemoryId(null)
   571→        }
   572→      }
   573→    } catch {
   574→      // Handle error silently
   575→    } finally {
   576→      setIsDeleting(false)
   577→      setDeleteTarget(null)
   578→    }
   579→  }, [deleteTarget, deleteMemory, selectedMemoryId, setSelectedMemoryId])
   580→
   581→  const handleBack = useCallback(() => {
   582→    setSelectedMemoryId(null)
   583→  }, [setSelectedMemoryId])
   584→
   585→  const hasFilters = searchQuery.trim() !== '' || filterType !== 'all' || !!selectedCollectionId
   586→
   587→  return (
   588→    <div className="max-w-3xl mx-auto pb-24 md:pb-8">
   589→      <AnimatePresence mode="wait">
   590→        {selectedMemory ? (
   591→          /* ── Detail View ──────────────────────────────────────────── */
   592→          <MemoryDetail
   593→            key="detail"
   594→            memory={selectedMemory}
   595→            onBack={handleBack}
   596→            onToggleFavorite={() => handleToggleFavorite(selectedMemory)}
   597→            onDelete={() => setDeleteTarget(selectedMemory.id)}
   598→          />
   599→        ) : (
   600→          /* ── List View ────────────────────────────────────────────── */
   601→          <motion.div
   602→            key="list"
   603→            variants={containerVariants}
   604→            initial="hidden"
   605→            animate="visible"
   606→            className="space-y-5"
   607→          >
   608→            {/* ── Header ────────────────────────────────────────────── */}
   609→            <motion.div variants={itemVariants}>
   610→              <div className="flex items-center justify-between">
   611→                <div>
   612→                  <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-foreground flex items-center gap-3">
