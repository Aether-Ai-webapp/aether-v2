    32→    32→function getGreeting(): string {
    33→    33→  const hour = new Date().getHours()
    34→    34→  if (hour < 12) return 'Good morning'
    35→    35→  if (hour < 17) return 'Good afternoon'
    36→    36→  return 'Good evening'

    39→    39→function detectContentType(text: string): 'link' | 'task' | 'note' {
    40→    40→  const lower = text.toLowerCase()
    41→    41→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    42→    42→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    43→    43→  return 'note'
    44→    44→}
    45→    45→
    46→    46→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    47→    47→  if (detected === 'link') return 'link'
    48→    48→  return 'text'
    49→    49→}

    70→    70→function generateParticles(count: number = 12): Particle[] {
    71→    71→  const colors = [
    72→    72→    'rgba(124, 58, 237, 0.8)',

    82→    82→    y: -(80 + Math.random() * 120),
    83→    83→    scale: 0.4 + Math.random() * 0.8,

   450→   450→    const detectedType = detectContentType(text)
