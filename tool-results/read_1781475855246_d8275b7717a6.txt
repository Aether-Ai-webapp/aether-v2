    79→function mapSupabaseMemory(row: SupabaseMemoryRow): Memory {
    80→  return {
    81→    id: row.id,
    82→    type: (row.type as MemoryType) || 'text',
    83→    title: row.title || '',
    84→    content: row.content || '',
    85→    summary: row.summary,
    86→    tags: row.tags ? row.tags.split(',').filter(Boolean) : [],
    87→    sourceUrl: row.source_url,
    88→    fileUrl: row.file_url,
    89→    imagePreview: row.image_preview,
    90→    imageUrl: row.image_url,

    93→    createdAt: row.created_at,

   115→// ─── Smart Token Saver: Local, free, instant tagging ──────────────────
   116→function getLocalTags(text: string): string[] {
   117→  const lower = text.toLowerCase()
   118→  const tags: string[] = []
   119→
   120→  // URL detection → 'link'
   121→  if (/https?:\/\//.test(lower)) {
   122→    tags.push('link')
   123→  }
   124→
   125→  // Task words → 'task'
   126→  if (/\b(todo|remind|need to|must|buy)\b/.test(lower)) {
   127→    tags.push('task')
   128→  }
   129→
   130→  // Person/personal words → 'personal'
   131→  if (/\b(i |me |with |call|email)\b/.test(lower)) {
   132→    tags.push('personal')
   133→  }
   134→
   135→  // Question/idea words → 'idea'
   136→  if (/\b(what|how|why|who)\b/.test(lower)) {
   137→    tags.push('idea')
   138→  }
   139→
   140→  return tags.slice(0, 5)
   141→}

   143→// ─── Extended keyword-based auto-tagging (richer, for AI fallback) ───
   144→function autoGenerateTags(content: string, title: string): string[] {
   145→  const text = `${title} ${content}`.toLowerCase()
   146→  const tagMap: Record<string, string[]> = {
   147→    'work': ['meeting', 'project', 'deadline', 'client', 'office', 'team', 'q1', 'q2', 'q3', 'q4', 'quarterly', 'strategy'],
   148→    'personal': ['routine', 'morning', 'exercise', 'meditation', 'journal', 'habit'],
   149→    'travel': ['trip', 'itinerary', 'flight', 'hotel', 'visit', 'tokyo', 'paris', 'destination'],
   150→    'learning': ['learn', 'study', 'course', 'tutorial', 'book', 'read', 'article'],
   151→    'code': ['code', 'programming', 'react', 'javascript', 'typescript', 'api', 'bug', 'feature', 'css', 'html', 'framework'],
   152→    'design': ['design', 'ui', 'ux', 'layout', 'color', 'font', 'figma', 'wireframe'],
   153→    'ai': ['ai', 'machine learning', 'neural', 'model', 'gpt', 'gemini', 'llm', 'chatbot'],
   154→    'recipe': ['recipe', 'cook', 'bake', 'ingredient', 'food', 'meal', 'breakfast', 'dinner'],
   155→    'idea': ['idea', 'concept', 'brainstorm', 'innovative', 'startup', 'prototype'],
   156→    'finance': ['budget', 'expense', 'invest', 'savings', 'money', 'cost'],
   157→  }
   158→
   159→  const tags: string[] = []
   160→  for (const [tag, keywords] of Object.entries(tagMap)) {
   161→    if (keywords.some(kw => text.includes(kw))) {
   162→      tags.push(tag)
   163→    }
   164→  }
   165→  return tags.slice(0, 5)
   166→}

   266→export const useAetherStore = create<AetherState>((set, get) => ({
   267→  // Navigation
   268→  currentView: 'dashboard',
   269→  setCurrentView: (view) => set({ currentView: view, selectedMemoryId: null, selectedCollectionId: null }),
   270→
   271→  // Auth — start as "local" user, no auth gate
   272→  user: { id: 'local', email: '', name: 'Aether User', avatarUrl: null },
   273→  isAuthenticated: false,
   274→  showAuthModal: false,
   275→  pendingAction: null,
   276→  supabaseReady: false,
   277→  setUser: (user) => set({ user, isAuthenticated: !!user && user.id !== 'local' }),
   278→  setShowAuthModal: (showAuthModal) => set({ showAuthModal }),
   279→
   280→  // The gate: if user is signed in, run action immediately. Otherwise, queue it and show modal.
   281→  requireAuth: (action) => {
   282→    const state = get()
   283→    if (state.isAuthenticated && state.user && state.user.id !== 'local') {
   284→      action()
   285→    } else {
   286→      set({ showAuthModal: true, pendingAction: action })
   287→    }
   288→  },
   289→
   290→  login: async (email, password) => {
   291→    try {
   292→      const res = await fetch('/api/auth/login', {
   293→        method: 'POST',
   294→        headers: { 'Content-Type': 'application/json' },
   295→        body: JSON.stringify({ email, password }),
   296→      })
   297→      if (!res.ok) return false
   298→      const data = await res.json()
   299→      if (data.user) {
   300→        set({
   301→          user: {
   302→            id: data.user.id,
   303→            email: data.user.email || '',
   304→            name: data.user.user_metadata?.name || '',
   305→            avatarUrl: data.user.user_metadata?.avatar_url || null,
   306→          },
   307→          isAuthenticated: true,
   308→          showAuthModal: false,
   309→        })
   310→        // CRITICAL: Await Supabase tables check before running pending actions or fetching data
   311→        // Without this, supabaseReady is still false when saveMemory/fetchMemories run
   312→        await get().checkSupabaseTables()
   313→        // Execute any pending action that was queued (e.g., deferred capture)
   314→        const pending = get().pendingAction
   315→        if (pending) {
   316→          await pending()
   317→          set({ pendingAction: null })
   318→        }
   319→        // Re-fetch data from Supabase (now that supabaseReady is set)
   320→        await get().fetchMemories()
   321→        await get().fetchCollections()
   322→        return true
   323→      }
   324→      return false
   325→    } catch {
   326→      return false
   327→    }
   328→  },
   329→
   330→  signup: async (email, password, name) => {
   331→    try {
   332→      const res = await fetch('/api/auth/signup', {
   333→        method: 'POST',
   334→        headers: { 'Content-Type': 'application/json' },
   335→        body: JSON.stringify({ email, password, name }),
   336→      })
   337→      if (!res.ok) return false
   338→      const data = await res.json()
   339→      if (data.user) {
   340→        set({
   341→          user: {
   342→            id: data.user.id,
   343→            email: data.user.email || '',
   344→            name: name || '',
   345→            avatarUrl: null,
   346→          },
   347→          isAuthenticated: true,
   348→          showAuthModal: false,
   349→        })
   350→        // CRITICAL: Await Supabase tables check before running pending actions or fetching data
   351→        await get().checkSupabaseTables()
   352→        // Execute any pending action
   353→        const pending = get().pendingAction
   354→        if (pending) {
   355→          await pending()
   356→          set({ pendingAction: null })
   357→        }
   358→        // Re-fetch data from Supabase (now that supabaseReady is set)
   359→        await get().fetchMemories()
   360→        await get().fetchCollections()
   361→        return true
   362→      }
   363→      return false
   364→    } catch {
   365→      return false
   366→    }
   367→  },
   368→
   369→  logout: async () => {
   370→    try {
   371→      await fetch('/api/auth/logout', { method: 'POST' })
   372→    } catch { /* ignore */ }
   373→    resetSupabaseBrowser()
   374→    set({
   375→      user: { id: 'local', email: '', name: 'Aether User', avatarUrl: null },
   376→      isAuthenticated: false,
   377→      supabaseReady: false,
   378→      chatMessages: [],
   379→    })
   380→    // Reload data as local user
   381→    get().fetchMemories()
   382→    get().fetchCollections()
   383→  },
   384→
   385→  checkSession: async () => {
   386→    try {
   387→      const res = await fetch('/api/auth/session')
   388→      if (res.ok) {
   389→        const data = await res.json()
   390→        if (data.authenticated && data.user) {
   391→          set({ user: data.user, isAuthenticated: true })
   392→          // CRITICAL: Await Supabase tables check, THEN fetch data
   393→          // Without this, fetchMemories runs while supabaseReady is still false
   394→          await get().checkSupabaseTables()
   395→          // Now fetch data with correct supabaseReady state
   396→          await Promise.all([
   397→            get().fetchMemories(),
   398→            get().fetchCollections(),
   399→          ])
   400→          return
   401→        }
   402→      }
   403→    } catch { /* ignore */ }
   404→    // No Supabase session — stay as local user (not gated)
   405→    set({ isAuthenticated: false })
   406→    // Still try to fetch from Prisma API for local data
   407→    await Promise.all([
   408→      get().fetchMemories(),
   409→      get().fetchCollections(),
   410→    ])
   411→  },
   412→
   413→  // Check if Supabase tables exist for the current user
   414→  checkSupabaseTables: async () => {
   415→    try {
   416→      const supabase = await getSupabaseBrowser()
   417→      const { data: { session } } = await supabase.auth.getSession()
   418→      if (!session) {
   419→        set({ supabaseReady: false })
   420→        return false
   421→      }
   422→
   423→      // Try a minimal query to see if the memories table exists AND RLS lets us read
   424→      const { error } = await supabase
   425→        .from('memories')
   426→        .select('id')
   427→        .eq('user_id', session.user.id)
   428→        .limit(1)
   429→
   430→      if (error) {
   431→        console.warn('[checkSupabaseTables] Probe failed:', error.message)
   432→        // Don't immediately give up — a permissions error on an empty table is normal
   433→        // If the error is specifically "no rows" that's fine, the table exists
   434→        // Only set supabaseReady=false for actual table-not-exist errors
   435→        const isTableMissing = error.message?.includes('does not exist') || error.message?.includes('not found')
   436→        if (!isTableMissing) {
   437→          // Table exists, we just can't read it (could be empty + RLS)
   438→          // This is actually fine — set ready=true
   439→          console.info('[checkSupabaseTables] Table exists but query returned error (likely empty table + RLS). Setting ready=true.')
   440→          set({ supabaseReady: true })
   441→          return true
   442→        }
   443→        set({ supabaseReady: false })
   444→        return false
   445→      }
   446→
   447→      set({ supabaseReady: true })
   448→      return true
   449→    } catch (err) {
   450→      console.error('[checkSupabaseTables] Exception:', err instanceof Error ? err.message : 'Unknown')
   451→      set({ supabaseReady: false })
   452→      return false
   453→    }
   454→  },
   455→
   456→  // Data
   457→  memories: [],
   458→  collections: [],
   459→  isLoading: false,
   460→
   461→  // Selected
   462→  selectedMemoryId: null,
   463→  selectedCollectionId: null,
   464→
   465→  // Search
   466→  searchQuery: '',
   467→  setSearchQuery: (q) => set({ searchQuery: q }),
   468→  filterType: 'all',
   469→  setFilterType: (t) => set({ filterType: t }),
   470→
   471→  // Chat
   472→  chatMessages: [],
   473→  addChatMessage: (msg) => set((s) => ({ chatMessages: [...s.chatMessages, msg] })),
   474→  clearChat: () => set({ chatMessages: [] }),
   475→
   476→  // Theme
   477→  darkMode: false,
   478→  toggleDarkMode: () => set((s) => ({ darkMode: !s.darkMode })),
   479→
   480→  // Data actions
   481→  setMemories: (memories) => set({ memories }),
   482→  setCollections: (collections) => set({ collections }),
   483→  setLoading: (isLoading) => set({ isLoading }),
   484→  setSelectedMemoryId: (id) => set({ selectedMemoryId: id }),
   485→  setSelectedCollectionId: (id) => set({ selectedCollectionId: id }),
   486→  addMemory: (memory) => set((s) => ({ memories: [memory, ...s.memories] })),
   487→  updateMemory: (id, updates) => set((s) => ({
   488→    memories: s.memories.map((m) => (m.id === id ? { ...m, ...updates } : m)),
   489→  })),
   490→  deleteMemory: (id) => set((s) => ({
   491→    memories: s.memories.filter((m) => m.id !== id),
   492→  })),
   493→  addCollection: (collection) => set((s) => ({
   494→    collections: [...s.collections, collection],
   495→  })),
   496→  updateCollection: (id, updates) => set((s) => ({
   497→    collections: s.collections.map((c) => (c.id === id ? { ...c, ...updates } : c)),
   498→  })),
   499→  deleteCollection: (id) => set((s) => ({
   500→    collections: s.collections.filter((c) => c.id !== id),
   501→  })),
   502→
   503→  // ── Fetch Memories ──────────────────────────────────────────────────
   504→  fetchMemories: async () => {
   505→    set({ isLoading: true })
   506→    const state = get()
   507→
   508→    // ── Authenticated + Supabase ready: fetch from Supabase directly ──
   509→    if (state.isAuthenticated && state.supabaseReady) {
   510→      try {
   511→        const supabase = await getSupabaseBrowser()
   512→        const { data: { session } } = await supabase.auth.getSession()
   513→        if (!session) throw new Error('No session')
   514→
   515→        const { data, error } = await supabase
   516→          .from('memories')
   517→          .select('*, memory_collections(collection_id, collections(id, name, color, icon))')
   518→          .eq('user_id', session.user.id)
   519→          .order('created_at', { ascending: false })
   520→
   521→        if (error) throw error
   522→
   523→        const memories = (data as SupabaseMemoryRow[]).map(mapSupabaseMemory)
   524→        set({ memories, isLoading: false })
   525→        return
   526→      } catch (supabaseErr) {
   527→        console.error('[fetchMemories] Supabase direct fetch failed:', supabaseErr instanceof Error ? supabaseErr.message : 'Unknown')
   528→        // Fall through to API
   529→      }
   530→    }
   531→
   532→    // ── Fallback: fetch from API (works for both auth and non-auth) ──
   533→    try {
   534→      const res = await fetch('/api/memories')
   535→      if (res.ok) {
   536→        const memories = await res.json()
   537→        set({ memories, isLoading: false })
   538→        return
   539→      }
   540→    } catch (e) {
   541→      console.error('[fetchMemories] API fetch failed:', e)
   542→    }
   543→    set({ isLoading: false })
   544→  },
   545→
   546→  // ── Fetch Collections ───────────────────────────────────────────────
   547→  fetchCollections: async () => {
   548→    const state = get()
   549→    if (state.isAuthenticated && state.supabaseReady) {
   550→      try {
   551→        const supabase = await getSupabaseBrowser()
   552→        const { data: { session } } = await supabase.auth.getSession()
   553→        if (!session) throw new Error('No session')
   554→
   555→        const { data, error } = await supabase
   556→          .from('collections')
   557→          .select('*, memory_collections(id)')
   558→          .eq('user_id', session.user.id)
   559→          .order('name', { ascending: true })
   560→
   561→        if (error) throw error
   562→
   563→        const collections = (data as SupabaseCollectionRow[]).map(mapSupabaseCollection)
   564→        set({ collections })
   565→        return
   566→      } catch {
   567→        // Fall through to Prisma API
   568→      }
   569→    }
   570→
   571→    // Fallback: fetch from Prisma API
   572→    try {
   573→      const res = await fetch('/api/collections')
   574→      if (res.ok) {
   575→        const collections = await res.json()
   576→        set({ collections })
   577→      }
   578→    } catch (e) {
   579→      console.error('Failed to fetch collections:', e)
   580→    }
   581→  },
   582→
   583→  // ── Delete Memory from DB (Supabase or Prisma) ─────────────────────
   584→  deleteMemoryFromDB: async (id: string) => {
   585→    const state = get()
   586→
   587→    // Authenticated + Supabase: delete from Supabase
   588→    if (state.isAuthenticated && state.supabaseReady) {
