   109→   109→// ─── Smart Token Saver: Local, free, instant tagging ──────────────────
   110→   110→function getLocalTags(text: string): string[] {
   111→   111→  const lower = text.toLowerCase()
   112→   112→  const tags: string[] = []
   113→   113→
   114→   114→  // URL detection → 'link'
   115→   115→  if (/https?:\/\//.test(lower)) {
   116→   116→    tags.push('link')
   117→   117→  }
   118→   118→
   119→   119→  // Task words → 'task'
   120→   120→  if (/\b(todo|remind|need to|must|buy)\b/.test(lower)) {
   121→   121→    tags.push('task')
   122→   122→  }
   123→   123→
   124→   124→  // Person/personal words → 'personal'
   125→   125→  if (/\b(i |me |with |call|email)\b/.test(lower)) {
   126→   126→    tags.push('personal')
   127→   127→  }
   128→   128→
   129→   129→  // Question/idea words → 'idea'
   130→   130→  if (/\b(what|how|why|who)\b/.test(lower)) {
   131→   131→    tags.push('idea')
   132→   132→  }
   133→   133→
   134→   134→  return tags.slice(0, 5)
   135→   135→}

   137→   137→// ─── Extended keyword-based auto-tagging (richer, for AI fallback) ───
   138→   138→function autoGenerateTags(content: string, title: string): string[] {
   139→   139→  const text = `${title} ${content}`.toLowerCase()
   140→   140→  const tagMap: Record<string, string[]> = {
   141→   141→    'work': ['meeting', 'project', 'deadline', 'client', 'office', 'team', 'q1', 'q2', 'q3', 'q4', 'quarterly', 'strategy'],
   142→   142→    'personal': ['routine', 'morning', 'exercise', 'meditation', 'journal', 'habit'],
   143→   143→    'travel': ['trip', 'itinerary', 'flight', 'hotel', 'visit', 'tokyo', 'paris', 'destination'],
   144→   144→    'learning': ['learn', 'study', 'course', 'tutorial', 'book', 'read', 'article'],
   145→   145→    'code': ['code', 'programming', 'react', 'javascript', 'typescript', 'api', 'bug', 'feature', 'css', 'html', 'framework'],
   146→   146→    'design': ['design', 'ui', 'ux', 'layout', 'color', 'font', 'figma', 'wireframe'],
   147→   147→    'ai': ['ai', 'machine learning', 'neural', 'model', 'gpt', 'gemini', 'llm', 'chatbot'],
   148→   148→    'recipe': ['recipe', 'cook', 'bake', 'ingredient', 'food', 'meal', 'breakfast', 'dinner'],
   149→   149→    'idea': ['idea', 'concept', 'brainstorm', 'innovative', 'startup', 'prototype'],
   150→   150→    'finance': ['budget', 'expense', 'invest', 'savings', 'money', 'cost'],
   151→   151→  }
   152→   152→
   153→   153→  const tags: string[] = []
   154→   154→  for (const [tag, keywords] of Object.entries(tagMap)) {
   155→   155→    if (keywords.some(kw => text.includes(kw))) {
   156→   156→      tags.push(tag)
   157→   157→    }
   158→   158→  }
   159→   159→  return tags.slice(0, 5)
   160→   160→}

   270→   270→
   271→   271→  // The gate: if user is signed in, run action immediately. Otherwise, queue it and show modal.
   272→   272→  requireAuth: (action) => {
   273→   273→    const state = get()
   274→   274→    if (state.isAuthenticated && state.user && state.user.id !== 'local') {
   275→   275→      action()

   277→   277→      set({ showAuthModal: true, pendingAction: action })
   278→   278→    }
   279→   279→  },
   280→   280→
   281→   281→  login: async (email, password) => {
   282→   282→    try {
   283→   283→      const res = await fetch('/api/auth/login', {
   284→   284→        method: 'POST',
   285→   285→        headers: { 'Content-Type': 'application/json' },
   286→   286→        body: JSON.stringify({ email, password }),
   287→   287→      })
   288→   288→      if (!res.ok) return false
   289→   289→      const data = await res.json()
   290→   290→      if (data.user) {
   291→   291→        set({
   292→   292→          user: {
   293→   293→            id: data.user.id,
   294→   294→            email: data.user.email || '',
   295→   295→            name: data.user.user_metadata?.name || '',
   296→   296→            avatarUrl: data.user.user_metadata?.avatar_url || null,
   297→   297→          },
   298→   298→          isAuthenticated: true,
   299→   299→          showAuthModal: false,
   300→   300→        })
   301→   301→        // CRITICAL: Await Supabase tables check before running pending actions or fetching data
   302→   302→        // Without this, supabaseReady is still false when saveMemory/fetchMemories run
   303→   303→        await get().checkSupabaseTables()
   304→   304→        // Execute any pending action that was queued (e.g., deferred capture)
   305→   305→        const pending = get().pendingAction
   306→   306→        if (pending) {
   307→   307→          await pending()
   308→   308→          set({ pendingAction: null })
   309→   309→        }
   310→   310→        // Re-fetch data from Supabase (now that supabaseReady is set)
   311→   311→        await get().fetchMemories()
   312→   312→        await get().fetchCollections()
   313→   313→        return true
   314→   314→      }
   315→   315→      return false
   316→   316→    } catch {
   317→   317→      return false
   318→   318→    }
   319→   319→  },
   320→   320→
   321→   321→  signup: async (email, password, name) => {
   322→   322→    try {
   323→   323→      const res = await fetch('/api/auth/signup', {
   324→   324→        method: 'POST',
   325→   325→        headers: { 'Content-Type': 'application/json' },

   327→   327→      })
   328→   328→      if (!res.ok) return false
   329→   329→      const data = await res.json()

   331→   331→        set({
   332→   332→          user: {
   333→   333→            id: data.user.id,
   334→   334→            email: data.user.email || '',
   335→   335→            name: name || '',
   336→   336→            avatarUrl: null,
   337→   337→          },
   338→   338→          isAuthenticated: true,
   339→   339→          showAuthModal: false,
   340→   340→        })
   341→   341→        // CRITICAL: Await Supabase tables check before running pending actions or fetching data

   343→   343→        // Execute any pending action
   344→   344→        const pending = get().pendingAction
   345→   345→        if (pending) {
   346→   346→          await pending()

   351→   351→        await get().fetchCollections()

   367→   367→      isAuthenticated: false,
   368→   368→      supabaseReady: false,

   379→   379→      if (res.ok) {
   380→   380→        const data = await res.json()
   381→   381→        if (data.authenticated && data.user) {
   382→   382→          set({ user: data.user, isAuthenticated: true })
   383→   383→          // CRITICAL: Await Supabase tables check, THEN fetch data
   384→   384→          // Without this, fetchMemories runs while supabaseReady is still false
   385→   385→          await get().checkSupabaseTables()

   387→   387→          await Promise.all([
   388→   388→            get().fetchMemories(),
   389→   389→            get().fetchCollections(),
   390→   390→          ])
   391→   391→          return
   392→   392→        }
   393→   393→      }
   394→   394→    } catch { /* ignore */ }
   395→   395→    // No Supabase session — stay as local user (not gated)
   396→   396→    set({ isAuthenticated: false })
   397→   397→    // Still try to fetch from Prisma API for local data
   398→   398→    await Promise.all([
   399→   399→      get().fetchMemories(),
   400→   400→      get().fetchCollections(),
   401→   401→    ])
   402→   402→  },
   403→   403→
   404→   404→  // Check if Supabase tables exist for the current user
   405→   405→  checkSupabaseTables: async () => {
   406→   406→    try {
   407→   407→      const supabase = await getSupabaseBrowser()
   408→   408→      const { data: { session } } = await supabase.auth.getSession()
   409→   409→      if (!session) {
   410→   410→        set({ supabaseReady: false })
   411→   411→        return false
   412→   412→      }
   413→   413→
   414→   414→      // Try a minimal query to see if the memories table exists AND RLS lets us read
   415→   415→      const { error } = await supabase
   416→   416→        .from('memories')
   417→   417→        .select('id')
   418→   418→        .eq('user_id', session.user.id)
   419→   419→        .limit(1)
   420→   420→
   421→   421→      if (error) {
   422→   422→        console.warn('[checkSupabaseTables] Probe failed:', error.message)
   423→   423→        // Don't immediately give up — a permissions error on an empty table is normal
   424→   424→        // If the error is specifically "no rows" that's fine, the table exists
   425→   425→        // Only set supabaseReady=false for actual table-not-exist errors
   426→   426→        const isTableMissing = error.message?.includes('does not exist') || error.message?.includes('not found')
   427→   427→        if (!isTableMissing) {
   428→   428→          // Table exists, we just can't read it (could be empty + RLS)
   429→   429→          // This is actually fine — set ready=true
   430→   430→          console.info('[checkSupabaseTables] Table exists but query returned error (likely empty table + RLS). Setting ready=true.')
   431→   431→          set({ supabaseReady: true })
   432→   432→          return true
   433→   433→        }
   434→   434→        set({ supabaseReady: false })
   435→   435→        return false
   436→   436→      }
   437→   437→
   438→   438→      set({ supabaseReady: true })
   439→   439→      return true
   440→   440→    } catch (err) {
   441→   441→      console.error('[checkSupabaseTables] Exception:', err instanceof Error ? err.message : 'Unknown')
   442→   442→      set({ supabaseReady: false })
   443→   443→      return false
   444→   444→    }
   445→   445→  },
