    32→function getGreeting(): string {
    33→  const hour = new Date().getHours()
    34→  if (hour < 12) return 'Good morning'
    35→  if (hour < 17) return 'Good afternoon'
    36→  return 'Good evening'
    37→}
    38→
    39→function detectContentType(text: string): 'link' | 'task' | 'note' {
    40→  const lower = text.toLowerCase()
    41→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    42→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    43→  return 'note'
    44→}
    45→
    46→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    47→  if (detected === 'link') return 'link'
    48→  return 'text'
    49→}

    70→function generateParticles(count: number = 12): Particle[] {
    71→  const colors = [
    72→    'rgba(124, 58, 237, 0.8)',
    73→    'rgba(99, 102, 241, 0.7)',
    74→    'rgba(79, 70, 229, 0.6)',
    75→    'rgba(59, 130, 246, 0.5)',
    76→    'rgba(37, 99, 235, 0.5)',
    77→    'rgba(236, 72, 153, 0.6)',
    78→  ]
    79→  return Array.from({ length: count }, (_, i) => ({
    80→    id: i,
    81→    x: (Math.random() - 0.5) * 160,
    82→    y: -(80 + Math.random() * 120),
    83→    scale: 0.4 + Math.random() * 0.8,
    84→    rotate: (Math.random() - 0.5) * 360,
    85→    duration: 0.5 + Math.random() * 0.5,
    86→    color: colors[Math.floor(Math.random() * colors.length)],
    87→    size: 4 + Math.random() * 8,
    88→  }))
    89→}

    91→function ParticleBurst({ particles, isDark }: { particles: Particle[]; isDark: boolean }) {
    92→  return (
    93→    <div className="absolute inset-0 pointer-events-none overflow-visible z-20">
    94→      {particles.map((p) => (
    95→        <motion.div
    96→          key={p.id}
    97→          initial={{ opacity: 1, x: 0, y: 0, scale: 0 }}
    98→          animate={{
    99→            opacity: [1, 1, 0],
   100→            x: p.x,
   101→            y: p.y,
   102→            scale: [0, p.scale, 0],
   103→            rotate: p.rotate,
   104→          }}
   105→          transition={{ duration: p.duration, ease: [0.22, 1, 0.36, 1] }}
   106→          className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 rounded-full"
   107→          style={{
   108→            width: p.size,
   109→            height: p.size,
   110→            backgroundColor: p.color,
   111→            boxShadow: isDark ? `0 0 ${p.size * 3}px ${p.color}` : `0 0 ${p.size}px ${p.color}`,
   112→          }}
   113→        />
   114→      ))}
   115→    </div>
   116→  )
   117→}

   262→
   263→  // ── Capture bar focus tracking
   264→  const [isCaptureFocused, setIsCaptureFocused] = useState(false)
   265→
   266→  const FREE_MEMORY_LIMIT = 15
   267→
   268→  // ── Upload image to Supabase Storage ────────────────────────────────
   269→  const uploadImageToStorage = useCallback(async (file: File): Promise<string | null> => {
   270→    try {
   271→      // Try Supabase Storage first
   272→      const { getSupabaseBrowser } = await import('@/lib/aether-store')
   273→      // We need the raw supabase client — access it via the store's internal helper
   274→      // Instead, use the browser client directly
   275→      const { createClient } = await import('@/lib/supabase/browser')
   276→      const supabase = createClient()
   277→
   278→      const fileExt = file.name.split('.').pop() || 'png'
   279→      const fileName = `memory-${Date.now()}-${Math.random().toString(36).slice(2, 8)}.${fileExt}`
   280→
   281→      const { error: uploadError } = await supabase.storage
   282→        .from('memory-assets')
   283→        .upload(fileName, file, { cacheControl: '3600', upsert: false })
   284→
   285→      if (uploadError) {
   286→        console.warn('[uploadImage] Supabase Storage upload failed:', uploadError.message)
   287→        // Return a data URL as fallback
   288→        return null
   289→      }
   290→
   291→      const { data: urlData } = supabase.storage
   292→        .from('memory-assets')
   293→        .getPublicUrl(fileName)
   294→
   295→      return urlData.publicUrl
   296→    } catch (err) {
   297→      console.warn('[uploadImage] Storage upload error:', err instanceof Error ? err.message : 'Unknown')
   298→      return null
   299→    }
   300→  }, [])
   301→
   302→  // ── Voice Recording handlers ────────────────────────────────────────
   303→  const startRecording = useCallback(async () => {
   304→    try {
   305→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   306→      const mediaRecorder = new MediaRecorder(stream, { mimeType: 'audio/webm;codecs=opus' })
   307→      mediaRecorderRef.current = mediaRecorder
   308→      audioChunksRef.current = []
   309→
   310→      mediaRecorder.ondataavailable = (event) => {
   311→        if (event.data.size > 0) {
   312→          audioChunksRef.current.push(event.data)
   313→        }
   314→      }
   315→
   316→      mediaRecorder.onstop = async () => {
   317→        stream.getTracks().forEach((track) => track.stop())
   318→        const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm;codecs=opus' })
   319→
   320→        // Send to Groq Whisper for transcription
   321→        setIsTranscribing(true)
   322→        try {
   323→          const formData = new FormData()
   324→          formData.append('audio', audioBlob, 'recording.webm')
   325→
   326→          const res = await fetch('/api/transcribe', {
   327→            method: 'POST',
   328→            body: formData,
   329→          })
   330→
   331→          if (res.ok) {
   332→            const { text } = await res.json()
   333→            if (text?.trim()) {
   334→              setCaptureText((prev) => prev ? `${prev} ${text}` : text)
   335→              toast.success('Voice transcribed!', { icon: <Mic className="size-4" /> })
   336→            } else {
   337→              toast.error('Could not detect speech. Try again.')
   338→            }
   339→          } else {
   340→            toast.error('Voice transcription failed.')
   341→          }
   342→        } catch (err) {
   343→          console.error('[transcribe] Error:', err)
   344→          toast.error('Voice transcription failed.')
   345→        } finally {
   346→          setIsTranscribing(false)
   347→        }
   348→      }
   349→
   350→      mediaRecorder.start()
   351→      setIsRecording(true)
   352→    } catch (err) {
   353→      console.error('[mic] Microphone access denied:', err)
   354→      toast.error('Microphone access denied. Please allow microphone permissions.')
   355→    }
   356→  }, [])
   357→
   358→  const stopRecording = useCallback(() => {
   359→    if (mediaRecorderRef.current && isRecording) {
   360→      mediaRecorderRef.current.stop()
   361→      setIsRecording(false)
   362→    }

   372→      toast.error('Please select an image file')

   439→            toast.error('Failed to save. Please try again.')
   440→          }
   441→        } catch {
   442→          toast.error('Something went wrong while saving.')
   443→        } finally {
   444→          setIsSaving(false)
   445→        }
   446→      })
   447→      return
   448→    }
   449→
   450→    const detectedType = detectContentType(text)
   451→    const memoryType = pendingImage ? 'image' : mapToMemoryType(detectedType)
   452→
   453→    setCaptureText('')
   454→    setParticles(generateParticles(14))
   455→    setTimeout(() => setParticles([]), 1200)
   456→
   457→    setCaptureFeedback({ category: detectedType })
   458→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   459→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   460→
   461→    setIsJustSaved(true)
   462→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   463→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)

   465→    if (memories.length >= FREE_MEMORY_LIMIT) {
   466→      setTimeout(() => setShowPaywall(true), 800)
   467→      return
   468→    }
   469→
   470→    // Upload image if present
   471→    let imageUrl: string | null = null
   472→    if (pendingImage) {
   473→      imageUrl = await uploadImageToStorage(pendingImage)
   474→      removePendingImage()
   475→    }
   476→
   477→    setIsSaving(true)
   478→    try {
   479→      const savedMemory = await saveMemory({
   480→        type: memoryType,
   481→        title: text.split('\n')[0].slice(0, 80) || (imageUrl ? 'Image note' : 'Quick Note'),
   482→        content: text,
   483→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   484→        imageUrl,
   485→      })
   486→
   487→      if (savedMemory) {
   488→        setLastSavedId(savedMemory.id)
   489→        setTimeout(() => setLastSavedId(null), 1500)
   490→        fetchMemories()
   491→      } else {
   492→        toast.error('Failed to save. Please try again.')
   493→      }
   494→    } catch {
   495→      toast.error('Something went wrong while saving.')
   496→    } finally {
   497→      setIsSaving(false)
   498→    }
   499→  }, [captureText, isSaving, saveMemory, fetchMemories, memories.length, FREE_MEMORY_LIMIT, isAuthenticated, requireAuth, pendingImage, uploadImageToStorage, removePendingImage])
   500→
   501→  const handleCaptureKeyDown = useCallback((e: React.KeyboardEvent) => {

   504→      handleCapture()

   534→      className={cn(
   535→        'min-h-screen relative overflow-hidden flex flex-col items-center pt-6 md:pt-10 pb-24 px-4',
   536→        isDark ? 'bg-transparent text-zinc-100' : 'bg-gradient-to-b from-slate-50 to-white text-gray-900'
   537→      )}
   538→    >
   539→      {/* Hidden file input for image upload */}
   540→      <input

   542→        type="file"

   548→      {/* ── Hero Greeting + Stats ──────────────────────────────────────── */}
