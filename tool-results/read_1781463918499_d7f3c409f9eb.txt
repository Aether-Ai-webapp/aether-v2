    28→// ─── Helpers ────────────────────────────────────────────────────────
    29→function getGreeting(): string {
    30→  const hour = new Date().getHours()
    31→  if (hour < 12) return 'Good morning'
    32→  if (hour < 17) return 'Good afternoon'
    33→  return 'Good evening'
    34→}
    35→
    36→function detectContentType(text: string): 'link' | 'task' | 'note' {
    37→  const lower = text.toLowerCase()
    38→  if (/https?:\/\//.test(lower) || /\bwww\./.test(lower)) return 'link'
    39→  if (/\b(todo|remind|need to|buy|must)\b/.test(lower)) return 'task'
    40→  return 'note'
    41→}
    42→
    43→function mapToMemoryType(detected: 'link' | 'task' | 'note'): MemoryType {
    44→  if (detected === 'link') return 'link'
    45→  return 'text'
    46→}

    55→// ─── Particle Burst Engine ───────────────────────────────────────────
    56→interface Particle {
    57→  id: number
    58→  x: number
    59→  y: number
    60→  scale: number
    61→  rotate: number
    62→  duration: number
    63→  color: string
    64→  size: number

    67→function generateParticles(count: number = 12): Particle[] {
    68→  const colors = [
    69→    'rgba(124, 58, 237, 0.8)',    // violet-600
    70→    'rgba(99, 102, 241, 0.7)',    // indigo-500
    71→    'rgba(79, 70, 229, 0.6)',     // indigo-600
    72→    'rgba(59, 130, 246, 0.5)',    // blue-500
    73→    'rgba(37, 99, 235, 0.5)',     // blue-600
    74→    'rgba(236, 72, 153, 0.6)',    // pink-500
    75→  ]
    76→  return Array.from({ length: count }, (_, i) => ({
    77→    id: i,
    78→    x: (Math.random() - 0.5) * 160,
    79→    y: -(80 + Math.random() * 120),
    80→    scale: 0.4 + Math.random() * 0.8,
    81→    rotate: (Math.random() - 0.5) * 360,
    82→    duration: 0.5 + Math.random() * 0.5,
    83→    color: colors[Math.floor(Math.random() * colors.length)],
    84→    size: 4 + Math.random() * 8,
    85→  }))
    86→}

    88→function ParticleBurst({ particles, isDark }: { particles: Particle[]; isDark: boolean }) {
    89→  return (
    90→    <div className="absolute inset-0 pointer-events-none overflow-visible z-20">
    91→      {particles.map((p) => (
    92→        <motion.div
    93→          key={p.id}
    94→          initial={{ opacity: 1, x: 0, y: 0, scale: 0 }}
    95→          animate={{
    96→            opacity: [1, 1, 0],
    97→            x: p.x,
    98→            y: p.y,
    99→            scale: [0, p.scale, 0],
   100→            rotate: p.rotate,
   101→          }}
   102→          transition={{
   103→            duration: p.duration,
   104→            ease: [0.22, 1, 0.36, 1],
   105→          }}
   106→          className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 rounded-full"
   107→          style={{
   108→            width: p.size,
   109→            height: p.size,
   110→            backgroundColor: p.color,
   111→            boxShadow: isDark
   112→              ? `0 0 ${p.size * 3}px ${p.color}`
   113→              : `0 0 ${p.size}px ${p.color}`,
   114→          }}
   115→        />
   116→      ))}
   117→    </div>
   118→  )
   119→}

   265→
   266→    // ── AUTH GATE: Block submission if not signed in ──────────────────

   268→      // Save the text so we can replay after login
   269→      const pendingText = text
   270→      setCaptureText('')
   271→      requireAuth(async () => {
   272→        // After successful auth, auto-capture the queued thought
   273→        const detectedType = detectContentType(pendingText)
   274→        const memoryType = mapToMemoryType(detectedType)
   275→
   276→        setParticles(generateParticles(14))
   277→        setTimeout(() => setParticles([]), 1200)

   282→        if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   283→        justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   284→
   285→        setIsSaving(true)
   286→        try {
   287→          const savedMemory = await saveMemory({
   288→            type: memoryType,
   289→            title: pendingText.split('\n')[0].slice(0, 80) || 'Quick Note',
   290→            content: pendingText,
   291→            sourceUrl: detectedType === 'link' ? pendingText.trim() : null,
   292→          })
   293→          if (savedMemory) {
   294→            setLastSavedId(savedMemory.id)
   295→            setTimeout(() => setLastSavedId(null), 1500)
   296→            // GUARANTEED HYDRATION after auth-gate replay
   297→            fetchMemories()
   298→          } else {
   299→            toast.error('Failed to save. Please try again.')
   300→          }
   301→        } catch {
   302→          toast.error('Something went wrong while saving.')
   303→        } finally {
   304→          setIsSaving(false)
   305→        }
   306→      })
   307→      return
   308→    }
   309→
   310→    const detectedType = detectContentType(text)
   311→    const memoryType = mapToMemoryType(detectedType)
   312→
   313→    setCaptureText('')
   314→    setParticles(generateParticles(14))
   315→    setTimeout(() => setParticles([]), 1200)
   316→
   317→    setCaptureFeedback({ category: detectedType })
   318→    if (feedbackTimer.current) clearTimeout(feedbackTimer.current)
   319→    feedbackTimer.current = setTimeout(() => setCaptureFeedback(null), 2500)
   320→
   321→    setIsJustSaved(true)
   322→    if (justSavedTimer.current) clearTimeout(justSavedTimer.current)
   323→    justSavedTimer.current = setTimeout(() => setIsJustSaved(false), 600)
   324→
   325→    if (memories.length >= FREE_MEMORY_LIMIT) {
   326→      setTimeout(() => setShowPaywall(true), 800)
   327→      return
   328→    }
   329→
   330→    setIsSaving(true)
   331→    try {
   332→      const savedMemory = await saveMemory({
   333→        type: memoryType,
   334→        title: text.split('\n')[0].slice(0, 80) || 'Quick Note',
   335→        content: text,
   336→        sourceUrl: detectedType === 'link' ? text.trim() : null,
   337→      })
   338→
   339→      if (savedMemory) {
   340→        setLastSavedId(savedMemory.id)
   341→        setTimeout(() => setLastSavedId(null), 1500)
   342→        // GUARANTEED HYDRATION: Re-fetch from backend to confirm persistence
   343→        fetchMemories()
   344→      } else {
   345→        toast.error('Failed to save. Please try again.')
   346→      }
   347→    } catch {
   348→      toast.error('Something went wrong while saving.')
   349→    } finally {
   350→      setIsSaving(false)
   351→    }
   352→  }, [captureText, isSaving, saveMemory, fetchMemories, memories.length, FREE_MEMORY_LIMIT, isAuthenticated, requireAuth])

   368→    try {
   369→      await deleteMemoryFromDB(selectedMemory.id)
